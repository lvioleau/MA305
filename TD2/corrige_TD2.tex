\begin{exo}{Exercice 1 -- Normes d'opérateur des matrices}
Soit $A=(a_{ij})\in M_d(\K)$, identifiée à l'endomorphisme $x\mapsto Ax$ de $\K^d$. Pour une norme $\norm\cdot$ sur $\K^d$, on note $\opnorm A$ la norme d'opérateur associée (même norme au départ et à l'arrivée).
\begin{enumerate}[label=(\alph*)]
  \item Montrer que pour $\norm\cdot_\infty$, $\opnorm A_\infty=\max_i\sum_j|a_{ij}|$ (plus grande somme en ligne).
  \item Montrer que pour $\norm\cdot_1$, $\opnorm A_1=\max_j\sum_i|a_{ij}|$ (plus grande somme en colonne).
  \item Montrer que toute norme d'opérateur vérifie $\opnorm{AB}\le\opnorm A\,\opnorm B$, et que toute valeur propre $\lambda$ de $A$ vérifie $|\lambda|\le\opnorm A$. Application : encadrer les valeurs propres de $\begin{pmatrix}1&2\\3&4\end{pmatrix}$.
  \item Montrer que pour la norme euclidienne $\norm\cdot_2$, $\opnorm A_2\le\big(\sum_{i,j}|a_{ij}|^2\big)^{1/2}$, et que l'inégalité peut être stricte.
\end{enumerate}
\end{exo}

\begin{solution}
\noindent\textbf{(a)} Notons $L:=\max_i\sum_j|a_{ij}|$. \emph{Majoration :} pour tout $i$,
\[
|(Ax)_i|\le\sum_j|a_{ij}|\,|x_j|\le\Big(\sum_j|a_{ij}|\Big)\norm x_\infty\le L\norm x_\infty .
\]
\emph{Égalité :} soit $i_0$ une ligne réalisant le maximum, et $x$ défini par $x_j:=\overline{a_{i_0j}}/|a_{i_0j}|$ si $a_{i_0j}\ne0$, $x_j:=1$ sinon. Alors $\norm x_\infty=1$ et $(Ax)_{i_0}=\sum_j|a_{i_0j}|=L$, donc $\norm{Ax}_\infty\ge L$.

\medskip
\noindent\textbf{(b)} Notons $C:=\max_j\sum_i|a_{ij}|$. \emph{Majoration :} en échangeant les deux sommes finies,
\[
\norm{Ax}_1\le\sum_i\sum_j|a_{ij}|\,|x_j|=\sum_j|x_j|\sum_i|a_{ij}|\le C\norm x_1 .
\]
\emph{Égalité :} si la colonne $j_0$ réalise le maximum, $\norm{e_{j_0}}_1=1$ et $Ae_{j_0}$ est la colonne $j_0$ de $A$, de norme $C$.

\medskip
\noindent\textbf{(c)} $\norm{ABx}\le\opnorm A\norm{Bx}\le\opnorm A\opnorm B\norm x$. Si $Ax=\lambda x$ avec $x\ne0$, alors $|\lambda|\norm x=\norm{Ax}\le\opnorm A\norm x$. Pour la matrice proposée, $\opnorm A_\infty=\max(3,7)=7$ et $\opnorm A_1=\max(4,6)=6$ : les valeurs propres sont de module $\le6$ (elles valent $\frac{5\pm\sqrt{33}}2$, soit environ $5{,}37$ et $-0{,}37$). On choisit la norme qui donne la meilleure borne.

\medskip
\noindent\textbf{(d)} Par Cauchy--Schwarz sur chaque ligne, $|(Ax)_i|^2\le\big(\sum_j|a_{ij}|^2\big)\norm x_2^2$, et en sommant sur $i$ : $\norm{Ax}_2^2\le\big(\sum_{i,j}|a_{ij}|^2\big)\norm x_2^2$. Pour $A=\mathrm{Id}$, $\opnorm{\mathrm{Id}}_2=1$ alors que le majorant vaut $\sqrt d$ : inégalité stricte dès que $d\ge2$.
\end{solution}

\begin{methode}
\textbf{Calculer une norme d'opérateur} se fait toujours en deux temps : (1) \emph{majorer} $\norm{Tx}\le C\norm x$, ce qui donne $\opnorm T\le C$ ; (2) \emph{saturer}, en exhibant un vecteur unitaire, ou une suite de vecteurs unitaires, pour lequel $\norm{Tx}$ atteint ou approche $C$. Le vecteur saturant se lit sur la majoration : il doit rendre chaque inégalité utilisée une égalité, ici signes alignés sur la ligne $i_0$ en (a), toute la masse sur la colonne $j_0$ en (b). Une majoration sans saturation, comme (d), ne donne qu'une borne.
\end{methode}

\begin{exo}{Exercice 2 -- Norme d'un opérateur de multiplication et d'un opérateur intégral}
\begin{enumerate}[label=(\alph*)]
  \item Soit $a\in\ell^\infty(\N)$ et $1\le p\le\infty$. Montrer que $M_a:u\mapsto(a_nu_n)_n$ est un opérateur de $\ell^p(\N)$ et que $\opnorm{M_a}=\norm a_\infty$.
  \item Pour $p<\infty$, montrer que la norme de $M_a$ est atteinte si et seulement si $\sup_n|a_n|$ est atteint. Que se passe-t-il pour $p=\infty$ ?
  \item Soit $k\in\Cz([0,1]^2;\R)$ et $Tf(x):=\int_0^1k(x,y)f(y)\,dy$ pour $f\in E:=\Cz([0,1];\R)$ muni de la norme uniforme. Montrer que $T\in\Lc(E)$ et que $\opnorm T\le M:=\max_{x\in[0,1]}\int_0^1|k(x,y)|\,dy$.
  \item Montrer que $\opnorm T=M$. \emph{Indication : si $x_0$ réalise le maximum, tester $f_\varepsilon(y):=\dfrac{k(x_0,y)}{|k(x_0,y)|+\varepsilon}$.}
\end{enumerate}
\end{exo}

\begin{solution}
\noindent\textbf{(a)} $|a_nu_n|\le\norm a_\infty|u_n|$, donc $M_au\in\ell^p$ et $\norm{M_au}_p\le\norm a_\infty\norm u_p$ : $M_a$ est linéaire continue, $\opnorm{M_a}\le\norm a_\infty$. Inversement $M_ae_n=a_ne_n$ avec $\norm{e_n}_p=1$, donc $\opnorm{M_a}\ge|a_n|$ pour tout $n$, d'où $\opnorm{M_a}\ge\sup_n|a_n|$.

\medskip
\noindent\textbf{(b)} Soit $s:=\norm a_\infty$ ; on suppose $a\ne0$. Si $|a_{n_0}|=s$, alors $\norm{M_ae_{n_0}}_p=s$ : atteinte. Réciproquement, supposons $|a_n|<s$ pour tout $n$ et soit $u$ unitaire. Il existe $n_1$ avec $u_{n_1}\ne0$, et
\[
\norm{M_au}_p^p=\sum_n|a_n|^p|u_n|^p<s^p\sum_n|u_n|^p=s^p,
\]
l'inégalité étant stricte à cause du terme $n_1$. La norme n'est donc pas atteinte.

Pour $p=\infty$ elle l'est toujours : avec $u=\mathbf 1$ (suite constante), $\norm{M_a\mathbf 1}_\infty=\sup_n|a_n|=s$.

\medskip
\noindent\textbf{(c)} $k$ est uniformément continue sur le compact $[0,1]^2$ (Théorème~1.3). Donc $|Tf(x)-Tf(x')|\le\norm f_\infty\sup_y|k(x,y)-k(x',y)|\to0$ quand $x'\to x$ : $Tf$ est continue, et $T$ est linéaire. Le même argument montre que $m:x\mapsto\int_0^1|k(x,y)|\,dy$ est continue, donc atteint son maximum $M$ sur $[0,1]$. Enfin
\[
|Tf(x)|\le\int_0^1|k(x,y)|\,|f(y)|\,dy\le m(x)\norm f_\infty\le M\norm f_\infty .
\]

\medskip
\noindent\textbf{(d)} La fonction $f_\varepsilon$ est continue (dénominateur $\ge\varepsilon>0$) et $\norm{f_\varepsilon}_\infty\le1$. En écrivant $\frac{k^2}{|k|+\varepsilon}=|k|-\frac{\varepsilon|k|}{|k|+\varepsilon}\ge|k|-\varepsilon$,
\[
Tf_\varepsilon(x_0)=\int_0^1\frac{k(x_0,y)^2}{|k(x_0,y)|+\varepsilon}\,dy\ge\int_0^1|k(x_0,y)|\,dy-\varepsilon=M-\varepsilon .
\]
Donc $\opnorm T\ge\norm{Tf_\varepsilon}_\infty\ge M-\varepsilon$ pour tout $\varepsilon>0$, et $\opnorm T=M$.
\end{solution}

\begin{piege}
\textbf{Le vecteur idéal n'est pas toujours dans l'espace.} En (d), la fonction qui saturerait la majoration est $\sgn k(x_0,\cdot)$ ; elle est discontinue dès que $k(x_0,\cdot)$ change de signe, donc hors de $\Cz$. On la \emph{régularise} en $f_\varepsilon$, et la norme est un supremum, pas forcément un maximum (elle est atteinte, par $f=\mathbf 1$, si $k\ge0$). C'est le même phénomène qu'à l'exercice officiel~7 pour le noyau de Dirichlet, et qu'en (b) : dans $\ell^p$, $p<\infty$, la masse ne peut pas se placer « à l'infini ». La formule de (d) est l'analogue continu de la norme $\opnorm A_\infty$ de l'Exercice~1(a).
\end{piege}

\begin{exo}{Exercice 3 -- Formes linéaires et hyperplans}
Soit $E$ un EVN, $f:E\to\K$ une forme linéaire non nulle et $H:=\Ker f$.
\begin{enumerate}[label=(\alph*)]
  \item Montrer que $H$ est soit fermé, soit dense dans $E$.
  \item Montrer que $f$ est continue si et seulement si $H$ est fermé.
  \item On suppose $f$ continue. Montrer que pour tout $x\in E$, $\dist(x,H)=\dfrac{|f(x)|}{\opnorm f}$.
  \item Montrer que, pour $x\notin H$, la distance $\dist(x,H)$ est atteinte si et seulement si $f$ atteint sa norme sur la sphère unité. Étudier l'exemple $E=c_0(\N)$, $f(u):=\sum_n2^{-n-1}u_n$.
\end{enumerate}
\end{exo}

\begin{solution}
\noindent\textbf{(a)} Fixons $a$ avec $f(a)=1$. Tout $x\in E$ s'écrit $x=(x-f(x)a)+f(x)a$ avec $x-f(x)a\in H$ : $E=H\oplus\K a$. Par conséquent un sous-espace vectoriel $V$ contenant strictement $H$ est égal à $E$ : il contient un $v=h+\lambda a$ avec $\lambda\ne0$, donc $a$, donc $E$. Or $\adh H$ est un sous-espace vectoriel (passage à la limite dans les combinaisons linéaires) contenant $H$ : ou bien $\adh H=H$, ou bien $\adh H=E$.

\medskip
\noindent\textbf{(b)} Si $f$ est continue, $H=f^{-1}(\{0\})$ est fermé. Réciproquement, si $H$ est fermé, alors $a\notin H$ donne $d:=\dist(a,H)>0$ (Corollaire~1.1). Pour $x$ avec $f(x)\ne0$, le vecteur $a-\frac x{f(x)}$ est dans $H$, donc
\[
\Big\|\frac x{f(x)}\Big\|=\Big\|a-\Big(a-\frac x{f(x)}\Big)\Big\|\ge d,\qquad\text{soit}\qquad|f(x)|\le\frac1d\norm x ,
\]
et $f$ est continue (Théorème~2.4). Avec (a) : \emph{le noyau d'une forme linéaire discontinue est dense}.

\medskip
\noindent\textbf{(c)} \emph{Minoration.} Pour $h\in H$, $|f(x)|=|f(x-h)|\le\opnorm f\norm{x-h}$ ; en prenant la borne inférieure sur $h$, $\dist(x,H)\ge|f(x)|/\opnorm f$.

\emph{Majoration.} Soit $\varepsilon\in\,]0,\opnorm f[$ et $u$ unitaire avec $|f(u)|\ge\opnorm f-\varepsilon$. Le vecteur $h:=x-\frac{f(x)}{f(u)}u$ est dans $H$ et $\norm{x-h}=\frac{|f(x)|}{|f(u)|}\le\frac{|f(x)|}{\opnorm f-\varepsilon}$. Donc $\dist(x,H)\le\frac{|f(x)|}{\opnorm f-\varepsilon}$ pour tout $\varepsilon$.

\medskip
\noindent\textbf{(d)} Si $u$ est unitaire avec $|f(u)|=\opnorm f$, le vecteur $h$ ci-dessus vérifie $\norm{x-h}=|f(x)|/\opnorm f=\dist(x,H)$ : la distance est atteinte. Réciproquement, si $h\in H$ vérifie $\norm{x-h}=\dist(x,H)=|f(x)|/\opnorm f$ avec $x\notin H$, alors $u:=\frac{x-h}{\norm{x-h}}$ est unitaire et $|f(u)|=\frac{|f(x)|}{\norm{x-h}}=\opnorm f$.

\emph{Exemple.} Sur $c_0$, $|f(u)|\le\sum_n2^{-n-1}|u_n|\le\norm u_\infty$, et $f(e_0+\dots+e_N)=1-2^{-N-1}\to1$ : $\opnorm f=1$. Mais si $\norm u_\infty\le1$, comme $u_n\to0$ on a $|u_n|<1$ à partir d'un certain rang, donc $|f(u)|<\sum_n2^{-n-1}=1$ : la norme n'est pas atteinte. Ainsi $H=\Ker f$ est un hyperplan \emph{fermé} de $c_0$ tel qu'\emph{aucun} point hors de $H$ n'a de plus proche point dans $H$.
\end{solution}

\begin{pointcle}
\textbf{Hyperplan fermé $\iff$ forme continue ; distance $=|f(x)|/\opnorm f$.} C'est la généralisation de la distance d'un point à un hyperplan de $\R^d$. À retenir aussi de (d) : dans un EVN quelconque, la distance à un sous-espace fermé n'est pas toujours atteinte. Elle l'est si le sous-espace est de dimension finie (Exercice~4), et dans un espace de Hilbert pour tout convexe fermé (Théorème~4.1) : ce sont deux mécanismes différents, la compacité et l'identité du parallélogramme.
\end{pointcle}

\begin{exo}{Exercice 4 -- Meilleure approximation dans un sous-espace de dimension finie}
Soit $E$ un EVN, $F$ un sous-espace de dimension finie et $x\in E$.
\begin{enumerate}[label=(\alph*)]
  \item Montrer qu'il existe $y\in F$ tel que $\norm{x-y}=\dist(x,F)$.
  \item En déduire que $F$ est fermé dans $E$.
  \item Montrer par un exemple dans $(\R^2,\norm\cdot_\infty)$ que $y$ n'est pas unique en général. Montrer qu'il est unique si la norme est \emph{strictement convexe} : $\norm u=\norm v=1$ et $u\ne v$ entraînent $\norm{\frac{u+v}2}<1$.
  \item Soit $f\in\Cz([0,1];\R)$ et $n\in\N$. Montrer qu'il existe un polynôme de degré $\le n$ réalisant la meilleure approximation uniforme de $f$, et que l'erreur $E_n(f):=\dist\big(f,\R_n[x]\big)$ décroît vers $0$.
\end{enumerate}
\end{exo}

\begin{solution}
\noindent\textbf{(a)} Soit $d:=\dist(x,F)$ et $(y_k)$ une suite de $F$ avec $\norm{x-y_k}\to d$ (suite minimisante). Elle est bornée : $\norm{y_k}\le\norm x+\norm{x-y_k}\le\norm x+d+1$ pour $k$ grand. Or $F$, muni de la norme induite, est un EVN de dimension finie, dans lequel les fermés bornés sont compacts (Corollaire~2.1) : une sous-suite $(y_{\varphi(k)})$ converge vers un $y\in F$. Par continuité de la norme, $\norm{x-y}=\lim\norm{x-y_{\varphi(k)}}=d$.

\medskip
\noindent\textbf{(b)} Soit $x\in\adh F$ : alors $\dist(x,F)=0$, et (a) fournit $y\in F$ avec $\norm{x-y}=0$, soit $x=y\in F$. (C'est la Proposition~3.2.)

\medskip
\noindent\textbf{(c)} Dans $(\R^2,\norm\cdot_\infty)$, soit $F:=\R(1,0)$ et $x:=(0,1)$. Pour $y=(t,0)$, $\norm{x-y}_\infty=\max(|t|,1)$, minimal et égal à $1$ pour tout $t\in[-1,1]$ : tout un segment de meilleures approximations.

Supposons la norme strictement convexe et soient $y_1\ne y_2$ deux meilleures approximations ; alors $d>0$ (sinon $x=y_1=y_2$). Les vecteurs $u:=\frac{x-y_1}d$ et $v:=\frac{x-y_2}d$ sont unitaires et distincts, donc $\norm{\frac{u+v}2}<1$, soit $\norm{x-\frac{y_1+y_2}2}<d$ avec $\frac{y_1+y_2}2\in F$ : contradiction avec la définition de $d$.

\medskip
\noindent\textbf{(d)} $\R_n[x]$ est un sous-espace de dimension $n+1$ de $(\Cz([0,1]),\norm\cdot_\infty)$ : (a) donne l'existence. La suite $(E_n(f))$ décroît car $\R_n[x]\subset\R_{n+1}[x]$. Par le théorème de Weierstrass (Exemple~2.3), pour tout $\varepsilon>0$ il existe un polynôme $P$ avec $\norm{f-P}_\infty\le\varepsilon$, donc $E_n(f)\le\varepsilon$ dès que $n\ge\deg P$.
\end{solution}

\begin{commentaire}
\textbf{Suite minimisante $+$ compacité.} C'est le schéma d'existence le plus simple en analyse : prendre une suite minimisante, montrer qu'elle est bornée, extraire, passer à la limite. Ici la compacité vient de la dimension finie de $F$, pas de celle de $E$. En dimension infinie le schéma échoue en général (Exercice~3(d)) ; dans un Hilbert on le sauve en montrant que la suite minimisante est de Cauchy grâce à l'identité du parallélogramme (Théorème~4.1), et au chapitre~5 en utilisant la compacité faible. La norme uniforme n'est pas strictement convexe, mais le polynôme de meilleure approximation de (d) est tout de même unique (théorème de Tchebychev, hors programme).
\end{commentaire}

\begin{exo}{Exercice 5 -- L'espace $c$ des suites convergentes}
On note $c$ l'ensemble des suites $u\in\K^\N$ convergentes, muni de $\norm\cdot_\infty$, et $\mathbf 1$ la suite constante égale à $1$.
\begin{enumerate}[label=(\alph*)]
  \item Montrer que $c$ est un sous-espace fermé de $\ell^\infty(\N)$.
  \item Montrer que $L:u\mapsto\lim_nu_n$ est une forme linéaire continue sur $c$, calculer sa norme, et montrer que $c=c_0(\N)\oplus\K\mathbf 1$.
  \item Montrer que $c_0(\N)$ et $c$ sont séparables.
  \item Soit $\Phi:c\to c_0(\N)$ définie par $\Phi(u):=\big(L(u),u_0-L(u),u_1-L(u),\dots\big)$. Montrer que $\Phi$ est linéaire, bijective, continue et d'inverse continu.
\end{enumerate}
\end{exo}

\begin{solution}
\noindent\textbf{(a)} $c$ est un sous-espace de $\ell^\infty$ (une suite convergente est bornée ; linéarité de la limite). Soit $u^k\in c$ avec $\norm{u^k-u}_\infty\to0$, $u\in\ell^\infty$. Pour $\varepsilon>0$, fixons $k$ avec $\norm{u^k-u}_\infty\le\varepsilon$ ; la suite $u^k$ converge, donc est de Cauchy : il existe $N$ tel que $|u^k_n-u^k_m|\le\varepsilon$ pour $n,m\ge N$. Alors
\[
|u_n-u_m|\le|u_n-u^k_n|+|u^k_n-u^k_m|+|u^k_m-u_m|\le3\varepsilon .
\]
La suite $u$ est de Cauchy dans $\K$ complet : $u\in c$.

\medskip
\noindent\textbf{(b)} $L$ est linéaire et $|L(u)|=\lim|u_n|\le\norm u_\infty$, avec égalité pour $u=\mathbf 1$ : $\opnorm L=1$. Son noyau est $c_0$, hyperplan fermé de $c$, et $L(\mathbf 1)=1$ : par l'Exercice~3(a), $c=c_0\oplus\K\mathbf 1$, la décomposition étant $u=(u-L(u)\mathbf 1)+L(u)\mathbf 1$.

\medskip
\noindent\textbf{(c)} Soit $D_0$ l'ensemble des suites à support fini et à coefficients dans $\Q$ (ou $\Q+i\Q$) : il est dénombrable. Pour $w\in c_0$ et $\varepsilon>0$, il existe $N$ avec $\sup_{n>N}|w_n|\le\varepsilon$ ; on approche $w_0,\dots,w_N$ à $\varepsilon$ près par des rationnels, ce qui donne $v\in D_0$ avec $\norm{w-v}_\infty\le\varepsilon$. Donc $c_0$ est séparable. Pour $c$ : $D:=\{v+q\mathbf 1:v\in D_0,\ q\in\Q\ (\text{ou }\Q+i\Q)\}$ est dénombrable, et pour $u\in c$ on approche $u-L(u)\mathbf 1\in c_0$ par $v\in D_0$ et $L(u)$ par $q$ : $\norm{u-(v+q\mathbf 1)}_\infty\le2\varepsilon$. Le sous-espace fermé $c$ de $\ell^\infty$ est séparable, alors que $\ell^\infty$ ne l'est pas (exercice officiel~9).

\medskip
\noindent\textbf{(d)} $\Phi(u)$ est bien dans $c_0$ puisque $u_n-L(u)\to0$, et $\Phi$ est linéaire. \emph{Continue :} $|L(u)|\le\norm u_\infty$ et $|u_n-L(u)|\le2\norm u_\infty$, donc $\norm{\Phi(u)}_\infty\le2\norm u_\infty$. \emph{Bijective, d'inverse continu :} pour $v\in c_0$, posons $\Psi(v):=(v_{n+1}+v_0)_{n\in\N}$. Cette suite converge vers $v_0$, donc $\Psi(v)\in c$ avec $L(\Psi(v))=v_0$, et $\norm{\Psi(v)}_\infty\le2\norm v_\infty$. On vérifie $\Phi(\Psi(v))=(v_0,v_1,v_2,\dots)=v$ et $\Psi(\Phi(u))=\big((u_n-L(u))+L(u)\big)_n=u$.
\end{solution}

\begin{commentaire}
\textbf{Isomorphes mais pas isométriques.} (d) dit que $c$ et $c_0$ sont le même espace vectoriel topologique : mêmes suites convergentes, mêmes formes linéaires continues à transport près. Mais les constantes $2$ ne peuvent pas être ramenées à $1$ : on peut montrer qu'il n'existe pas d'isométrie linéaire de $c$ sur $c_0$ (la boule unité de $c$ a des points extrémaux, comme $\mathbf 1$, celle de $c_0$ n'en a aucun). À retenir pour la suite : la chaîne $c_{<\infty}\subset c_0\subset c\subset\ell^\infty$, où $c_0$ est l'adhérence de $c_{<\infty}$, et où seul le dernier espace n'est pas séparable.
\end{commentaire}

\begin{exo}{Exercice 6 -- Applications du théorème de Stone--Weierstrass}
\begin{enumerate}[label=(\alph*)]
  \item \emph{(Problème des moments.)} Soit $f\in\Cz([0,1];\R)$ telle que $\int_0^1f(x)x^n\,dx=0$ pour tout $n\in\N$. Montrer que $f=0$.
  \item Même question en supposant seulement $\int_0^1f(x)x^{2n}\,dx=0$ pour tout $n\in\N$. Le résultat subsiste-t-il sur $[-1,1]$ ?
  \item Montrer que les fonctions de la forme $(x,y)\mapsto\sum_{i=1}^Nu_i(x)v_i(y)$, avec $u_i,v_i\in\Cz([0,1];\R)$, sont denses dans $\Cz([0,1]^2;\R)$.
  \item Montrer par des exemples que chacune des deux hypothèses « unitaire » et « séparant les points » est nécessaire. Montrer que les polynômes en $z$ ne sont pas denses dans $\Cz(\Sun;\C)$.
\end{enumerate}
\end{exo}

\begin{solution}
\noindent\textbf{(a)} Par linéarité, $\int_0^1fP=0$ pour tout polynôme $P$. Par le théorème de Weierstrass (Exemple~2.3), il existe des polynômes $P_k\to f$ uniformément sur $[0,1]$. Alors
\[
\int_0^1f^2=\int_0^1f(f-P_k)\le\norm f_\infty\norm{f-P_k}_\infty\xrightarrow[k\to\infty]{}0 ,
\]
donc $\int_0^1f^2=0$ et, $f^2$ étant continue et positive, $f=0$.

\medskip
\noindent\textbf{(b)} Soit $\mathcal A$ l'ensemble des polynômes en $x^2$. C'est une sous-algèbre de $\Cz([0,1];\R)$, unitaire, et elle sépare les points de $[0,1]$ car $x\mapsto x^2$ y est injective. Par le Théorème~2.2 elle est dense, et l'argument de (a) s'applique : $f=0$.

Sur $[-1,1]$ c'est faux : $x^2$ ne sépare pas $x$ et $-x$. Toute fonction \emph{impaire} $f$, par exemple $f(x)=x$, vérifie
\[
\int_{-1}^1f(x)x^{2n}\,dx=0\qquad\text{pour tout }n\quad\text{(intégrande impair).}
\] L'adhérence de $\mathcal A$ dans $\Cz([-1,1])$ est l'espace des fonctions continues paires.

\medskip
\noindent\textbf{(c)} Ces fonctions forment un sous-espace vectoriel $\mathcal A$ de $\Cz([0,1]^2;\R)$, stable par produit (le produit de $u(x)v(y)$ et $u'(x)v'(y)$ est $(uu')(x)(vv')(y)$), contenant les constantes. Il sépare les points : si $(x,y)\ne(x',y')$, alors $x\ne x'$ et la fonction $(x,y)\mapsto x$ les distingue, ou $y\ne y'$ et $(x,y)\mapsto y$ convient. Le compact $[0,1]^2$ étant métrique, le Théorème~2.2 donne la densité.

\medskip
\noindent\textbf{(d)} \emph{Sans séparation :} $\mathcal A:=\{f\in\Cz([0,1]):f(0)=f(1)\}$ est une sous-algèbre unitaire, fermée (la condition passe à la limite uniforme) et stricte ; elle ne sépare pas $0$ et $1$. \emph{Sans unité :} $\mathcal A:=\{f:f(0)=0\}$ est une sous-algèbre fermée et stricte, qui sépare les points (elle contient $x\mapsto x$) mais pas les constantes non nulles.

\emph{Cas complexe.} Pour un polynôme $P(z)=\sum_ka_kz^k$, $\int_0^{2\pi}P(e^{i\theta})e^{i\theta}\,d\theta=\sum_ka_k\int_0^{2\pi}e^{i(k+1)\theta}\,d\theta=0$. Cette égalité passe aux limites uniformes. Or pour $f(z)=\bar z$, $\int_0^{2\pi}e^{-i\theta}e^{i\theta}\,d\theta=2\pi\ne0$ : $\bar z$ n'est pas limite uniforme de polynômes en $z$ sur $\Sun$. L'algèbre des polynômes en $z$ est unitaire et séparante, mais pas stable par conjugaison : dans le cas complexe cette hypothèse supplémentaire est indispensable (Exemple~2.4, où l'on prend les $z^n$ pour $n\in\Z$).
\end{solution}

\begin{methode}
\textbf{Le schéma « densité $+$ orthogonalité ».} Pour montrer qu'une fonction continue est nulle à partir d'une famille de conditions intégrales $\int f\,g=0$, $g\in\mathcal A$ : (1) montrer que $\mathcal A$ est dense pour la norme uniforme, par Stone--Weierstrass, en vérifiant \emph{algèbre, unitaire, séparante} (et stable par conjugaison dans le cas complexe) ; (2) approcher $f$ elle-même par $g_k\in\mathcal A$ et écrire $\int f^2=\int f(f-g_k)\to0$. C'est la version « fonctions continues » du critère hilbertien \emph{totale $\iff$ orthogonal nul} (Théorème~4.3), et c'est ainsi qu'on montre que les exponentielles forment une base hilbertienne de $L^2$ (Exemple~4.7).
\end{methode}

\begin{exo}{Exercice officiel 1 -- Pas de Banach à base dénombrable}
Montrer qu'un espace vectoriel possédant une base infinie dénombrable n'est complet pour aucune norme (on pourra admettre qu'un sous-espace de dimension finie est toujours fermé).
\end{exo}

\begin{solution}
Soit $E$ un espace vectoriel muni d'une base $(e_n)_{n\in\N}$ (base \emph{algébrique} : tout vecteur est combinaison linéaire \emph{finie} des $e_n$, de façon unique). Soit $\norm\cdot$ une norme sur $E$ ; supposons par l'absurde $(E,\norm\cdot)$ complet.

\emph{Une union dénombrable de fermés.} Posons $F_n:=\Vect(e_0,\dots,e_n)$. Chaque $F_n$ est un sous-espace de dimension finie $n+1$, donc fermé (Proposition~3.2 du poly, admise dans l'énoncé). Tout $x\in E$ étant combinaison linéaire finie des $e_k$, il appartient à $F_n$ dès que $n$ dépasse le plus grand indice intervenant : $E=\bigcup_{n\in\N}F_n$.

\emph{Baire.} L'espace $E$ étant complet, le Lemme~1.1 (ii) interdit que tous les $F_n$ soient d'intérieur vide : il existe $N$ et une boule $B(x_0,r)\subset F_N$, $r>0$.

\emph{Un sous-espace d'intérieur non vide est tout l'espace.} Pour $y\in E\setminus\{0\}$, le vecteur $v:=x_0+\frac{r}{2\norm y}y$ est dans $B(x_0,r)\subset F_N$, et $x_0\in F_N$, donc $y=\frac{2\norm y}{r}(v-x_0)\in F_N$ (sous-espace vectoriel). Ainsi $F_N=E$.

\emph{Contradiction.} $e_{N+1}\in E=F_N=\Vect(e_0,\dots,e_N)$ contredit la liberté de la famille $(e_n)$. Donc $(E,\norm\cdot)$ n'est pas complet, et ce quelle que soit la norme choisie.
\end{solution}

\begin{pointcle}
C'est exactement le schéma de Baire vu au TD1 (exercice officiel 7) : \emph{espace complet $=$ union dénombrable de fermés $\Rightarrow$ l'un d'eux a un intérieur non vide $\Rightarrow$ la structure linéaire propage cette information à tout l'espace}. Conséquence : \textbf{la dimension algébrique d'un espace de Banach est finie ou non dénombrable}. Par exemple $\R[X]$ (base $(X^n)_n$) et $c_{<\infty}(\N)$ (suites à support fini, base $(\mathbf e^k)_k$) ne sont complets pour aucune norme, ce qui répond à l'Exercice~3.2 du poly.
\end{pointcle}

\begin{piege}
\textbf{Base algébrique $\ne$ base hilbertienne.} L'espace $\ell^2(\N)$ est complet et admet la famille orthonormée $(\mathbf e^k)_k$ comme « base hilbertienne » (tout vecteur est une \emph{série} convergente $\sum u_k\mathbf e^k$). Ce n'est pas une contradiction : $(\mathbf e^k)$ n'engendre algébriquement que $c_{<\infty}(\N)\subsetneq\ell^2(\N)$. L'énoncé parle de combinaisons \emph{finies} ; c'est précisément ce qui donne $E=\bigcup F_n$.
\end{piege}

\begin{exo}{Exercice officiel 2 -- Distance associée à une norme ; compacts de $\R^d$}
\begin{enumerate}[label=(\alph*)]
  \item Montrer que si $(E,\norm\cdot)$ est un espace normé, l'application $(x,y)\mapsto\norm{x-y}$ est une distance.
  \item On munit $\R^d$ de la norme euclidienne $\norm x=(\sum_{i=1}^d|x_i|^2)^{1/2}$. Montrer que tout sous-ensemble fermé borné est compact.
\end{enumerate}
\end{exo}

\begin{solution}
\noindent\textbf{(a)} Posons $d(x,y):=\norm{x-y}$, à valeurs dans $\R_+$. On vérifie les trois axiomes à partir de ceux de la norme (Définition~2.1).
\begin{itemize}
  \item \emph{Séparation :} $d(x,y)=0\iff\norm{x-y}=0\iff x-y=0\iff x=y$.
  \item \emph{Symétrie :} $d(y,x)=\norm{y-x}=\norm{(-1)(x-y)}=|-1|\,\norm{x-y}=d(x,y)$ par homogénéité.
  \item \emph{Inégalité triangulaire :} pour $x,y,z\in E$, en écrivant $x-z=(x-y)+(y-z)$,
  \[
  d(x,z)=\norm{(x-y)+(y-z)}\le\norm{x-y}+\norm{y-z}=d(x,y)+d(y,z).
  \]
\end{itemize}
Cette distance a deux propriétés supplémentaires qui la distinguent d'une distance quelconque : elle est \emph{invariante par translation}, $d(x+a,y+a)=d(x,y)$, et \emph{homogène}, $d(\lambda x,\lambda y)=|\lambda|d(x,y)$.

\medskip
\noindent\textbf{(b)} Soit $A\subset\R^d$ fermé et borné, et $(x^{(k)})_{k\in\N}$ une suite d'éléments de $A$. Il faut en extraire une sous-suite convergeant vers un élément de $A$ (Définition~1.20). On note $x^{(k)}=(x^{(k)}_1,\dots,x^{(k)}_d)$.

\emph{Étape 1 : chaque coordonnée est bornée.} $A$ borné signifie qu'il existe $R$ tel que $\norm{x}\le R$ pour tout $x\in A$. Comme $|x_i|\le\big(\sum_j|x_j|^2\big)^{1/2}=\norm x$, on a $x^{(k)}_i\in[-R,R]$ pour tous $k$ et $i$.

\emph{Étape 2 : extractions successives.} Par le théorème de Bolzano--Weierstrass (Théorème~1.2), la suite réelle bornée $(x^{(k)}_1)_k$ admet une sous-suite convergente : il existe une extractrice $\varphi_1$ et $\ell_1\in[-R,R]$ avec $x^{(\varphi_1(k))}_1\to\ell_1$. La suite $(x^{(\varphi_1(k))}_2)_k$ est encore bornée : on en extrait une sous-suite convergente via $\varphi_2$, $x^{(\varphi_1\circ\varphi_2(k))}_2\to\ell_2$ ; la première coordonnée converge toujours vers $\ell_1$ le long de $\varphi_1\circ\varphi_2$ (sous-suite d'une suite convergente). On itère $d$ fois et l'on pose $\varphi:=\varphi_1\circ\cdots\circ\varphi_d$ (extractrice comme composée d'extractrices) : pour chaque $i\in\{1,\dots,d\}$, $x^{(\varphi(k))}_i\to\ell_i$.

\emph{Étape 3 : la convergence coordonnée par coordonnée entraîne la convergence en norme.} Posons $\ell:=(\ell_1,\dots,\ell_d)$. Alors
\[
\norm{x^{(\varphi(k))}-\ell}^2=\sum_{i=1}^d\big|x^{(\varphi(k))}_i-\ell_i\big|^2\xrightarrow[k\to\infty]{}0
\]
comme somme \emph{finie} de suites tendant vers $0$. Donc $x^{(\varphi(k))}\to\ell$ dans $(\R^d,\norm\cdot)$.

\emph{Étape 4 : la limite est dans $A$.} $A$ est fermé et $(x^{(\varphi(k))})_k$ est une suite de $A$ convergeant vers $\ell$ : par la caractérisation séquentielle des fermés (Proposition~1.3), $\ell\in A$. Donc $A$ est compact.
\end{solution}

\begin{pointcle}
La preuve tient à deux faits : (i) dans $\R^d$, \emph{converger en norme, c'est converger coordonnée par coordonnée} (encadrement $\norm x_\infty\le\norm x_2\le\sqrt d\,\norm x_\infty$, cas particulier de (2.1)) ; (ii) on ne fait qu'un \emph{nombre fini} ($d$) d'extractions successives. C'est le Corollaire~1.2 du poly (prouvé là via la compacité du produit $[-R,R]^d$, Proposition~1.9), et le Théorème~2.1 l'étend à toute norme sur $\K^d$ puisqu'elles sont toutes équivalentes.
\end{pointcle}

\begin{piege}
L'argument s'effondre en dimension infinie : dans $\ell^2(\N)$, la suite $(\mathbf e^k)_k$ converge « coordonnée par coordonnée » vers $0$ (chaque coordonnée est nulle à partir d'un certain rang), mais $\norm{\mathbf e^k-\mathbf e^j}_2=\sqrt2$ pour $k\ne j$ : aucune sous-suite ne converge. Il y a une infinité de coordonnées, et « somme finie de suites tendant vers $0$ » n'a plus de sens. C'est le contenu du théorème de Riesz (Théorème~3.1) : \emph{fermé borné $\Rightarrow$ compact} caractérise la dimension finie.
\end{piege}

\begin{exo}{Exercice officiel 3 (Exercice 2.2 du poly) -- Suite exhaustive de compacts}
Soit $O$ un ouvert de $\R^d$, $F$ son complémentaire et $B_n$ la boule fermée de rayon $n$ centrée en l'origine. Montrer que les ensembles
\[
K_n:=B_n\cap\{x\in O : d(F,x)\ge1/n\}
\]
forment une suite croissante de compacts tous contenus dans $O$ et le recouvrant.
\end{exo}

\begin{solution}
Si $O=\R^d$, alors $F=\emptyset$ et, avec la convention $d(\emptyset,x)=\inf\emptyset=+\infty$, on a simplement $K_n=B_n$ : boules fermées, compactes, croissantes, recouvrant $\R^d$. On suppose désormais $F\ne\emptyset$. Rappelons que $F=\R^d\setminus O$ est fermé et que, par le Corollaire~1.1, $d(F,x)=0\iff x\in\adh F=F$ ; autrement dit
\begin{equation}\label{eq:dF}
x\in O\iff d(F,x)>0 .
\end{equation}

\emph{$K_n\subset O$.} C'est immédiat par définition ; on peut d'ailleurs, grâce à \eqref{eq:dF}, réécrire $K_n=B_n\cap\{x\in\R^d : d(F,x)\ge1/n\}$ sans mentionner $O$.

\emph{$K_n$ est compact.} La fonction $x\mapsto d(F,x)$ est $1$-lipschitzienne (TD1, exercice officiel~6), donc continue, et $\{x\in\R^d : d(F,x)\ge1/n\}$ est l'image réciproque du fermé $[1/n,+\infty[$ : c'est un fermé de $\R^d$. Ainsi $K_n$ est l'intersection de deux fermés, donc fermé, et il est borné car contenu dans $B_n$. Par le Corollaire~1.2 (ou l'exercice officiel~2), $K_n$ est compact.

\emph{Croissance.} Si $x\in K_n$, alors $\norm x\le n\le n+1$ et $d(F,x)\ge\frac1n\ge\frac1{n+1}$, donc $x\in K_{n+1}$.

\emph{Recouvrement.} Soit $x\in O$. Par \eqref{eq:dF}, $d(F,x)>0$. Choisissons un entier $n$ tel que $n\ge\norm x$ et $n\ge1/d(F,x)$ : alors $\norm x\le n$ et $d(F,x)\ge1/n$, \ie{} $x\in K_n$. Donc $O=\bigcup_nK_n$.

\medskip
\noindent\textbf{Complément utile : $K_n\subset\interieur{K_{n+1}}$.} Soit $x\in K_n$ et $\rho:=\min\big(1,\frac1n-\frac1{n+1}\big)>0$. Pour $y\in B(x,\rho)$, on a $\norm y\le\norm x+\rho<n+1$, et par le caractère $1$-lipschitzien de $d(F,\cdot)$,
\[
d(F,y)\ \ge\ d(F,x)-\norm{x-y}\ >\ \frac1n-\Big(\frac1n-\frac1{n+1}\Big)=\frac1{n+1}.
\]
Donc $B(x,\rho)\subset K_{n+1}$, \ie{} $x\in\interieur{K_{n+1}}$.

\emph{Conséquence : tout compact $K\subset O$ est contenu dans l'un des $K_n$.} Les ouverts $(\interieur{K_{n}})_{n\ge1}$ recouvrent $O\supset K$ (car $K_n\subset\interieur{K_{n+1}}$ et les $K_n$ recouvrent $O$). Par la propriété de Borel--Lebesgue (Théorème~1.4), $K$ est recouvert par un nombre fini d'entre eux, $K\subset\interieur{K_{n_1}}\cup\dots\cup\interieur{K_{n_p}}\subset K_{\max n_i}$ par croissance.
\end{solution}

\begin{pointcle}
Les deux contraintes qui définissent $K_n$ répondent aux deux façons pour un point de $O$ de « s'échapper » : partir à l'infini (contrôlé par $B_n$) ou s'approcher du bord $F=\partial O\cup(\R^d\setminus\adh O)$ (contrôlé par $d(F,x)\ge1/n$). Le complément $K_n\subset\interieur{K_{n+1}}$ est la propriété vraiment utile en pratique : c'est elle qui donne « tout compact de $O$ est dans un $K_n$ », indispensable à la question (c) de l'exercice suivant.
\end{pointcle}

\begin{exo}{Exercice officiel 4 -- La distance de la convergence uniforme sur tout compact}
Soit $O$ un ouvert de $\R^d$, $X:=\Cz(O)$ et $(K_n)_{n\ge0}$ une suite exhaustive de compacts associée à $O$. On pose
\[
d_X(f,g):=\sum_{n\ge0}\frac{\min\{1,\norm{f-g}_{K_n}\}}{2^n},\qquad f,g\in X .
\]
\begin{enumerate}[label=(\alph*)]
  \item Montrer que $d_X$ est une distance sur $X$.
  \item Soit $f\in X$ et $(f_j)_{j\in\N}\in X^\N$. Montrer que $\lim_jd_X(f_j,f)=0\iff f_j\to f$ uniformément sur $K_n$ pour tout $n\ge0$.
  \item Conclure que la topologie induite par $d_X$ ne dépend pas du choix de la suite $(K_n)_{n\ge0}$.
\end{enumerate}
\end{exo}

\begin{solution}
Remarquons d'abord que pour $f\in\Cz(O)$ et $K\subset O$ compact, $\norm f_K=\sup_K|f|<+\infty$ (fonction continue sur un compact, Corollaire~1.3) ; $\norm\cdot_{K}$ vérifie l'homogénéité et l'inégalité triangulaire, mais pas la séparation sur $\Cz(O)$ : c'est une \emph{semi-norme}.

\medskip
\noindent\textbf{(a)} \emph{$d_X$ est bien définie.} Le terme général de la série est compris entre $0$ et $2^{-n}$ : la série converge et $0\le d_X(f,g)\le2$.

\emph{Symétrie.} $\norm{f-g}_{K_n}=\norm{g-f}_{K_n}$.

\emph{Séparation.} Si $d_X(f,g)=0$, chaque terme (positif) de la série est nul : $\min\{1,\norm{f-g}_{K_n}\}=0$, donc $\norm{f-g}_{K_n}=0$, \ie{} $f=g$ sur $K_n$, pour tout $n$. Comme $\bigcup_nK_n=O$, $f=g$ sur $O$.

\emph{Inégalité triangulaire.} Le point crucial est que $\theta:t\mapsto\min\{1,t\}$ est \emph{croissante} et \emph{sous-additive} sur $\R_+$ :
\[
\theta(a+b)\le\theta(a)+\theta(b)\qquad(a,b\ge0).
\]
En effet, si $a\ge1$ ou $b\ge1$, le membre de droite vaut au moins $1\ge\theta(a+b)$ ; sinon il vaut $a+b\ge\theta(a+b)$. Pour $f,g,h\in X$ et tout $n$, l'inégalité triangulaire de la semi-norme $\norm\cdot_{K_n}$ puis la croissance et la sous-additivité de $\theta$ donnent
\[
\theta\big(\norm{f-h}_{K_n}\big)\le\theta\big(\norm{f-g}_{K_n}+\norm{g-h}_{K_n}\big)\le\theta\big(\norm{f-g}_{K_n}\big)+\theta\big(\norm{g-h}_{K_n}\big).
\]
On divise par $2^n$ et l'on somme (séries convergentes à termes positifs) : $d_X(f,h)\le d_X(f,g)+d_X(g,h)$.

\medskip
\noindent\textbf{(b)} $(\Rightarrow)$ Supposons $d_X(f_j,f)\to0$ et fixons $n$. Le terme d'indice $n$ est majoré par la somme :
\[
\min\{1,\norm{f_j-f}_{K_n}\}\le2^nd_X(f_j,f)\xrightarrow[j\to\infty]{}0 .
\]
Pour $j$ assez grand, ce minimum est $<1$, donc vaut $\norm{f_j-f}_{K_n}$, qui tend ainsi vers $0$ : $f_j\to f$ uniformément sur $K_n$.

$(\Leftarrow)$ Supposons $\norm{f_j-f}_{K_n}\to0$ pour tout $n$, et soit $\varepsilon>0$. \emph{On coupe la queue de la série :} choisissons $N$ tel que $\sum_{n>N}2^{-n}=2^{-N}\le\varepsilon/2$. Pour les \emph{finitement nombreux} indices $n\le N$, il existe $J$ tel que $j\ge J\Rightarrow\norm{f_j-f}_{K_n}\le\varepsilon/4$ pour tout $n\le N$ (prendre le max des rangs obtenus pour chaque $n$). Alors, pour $j\ge J$,
\[
d_X(f_j,f)\le\sum_{n\le N}\frac{\varepsilon/4}{2^n}+\sum_{n>N}\frac1{2^n}\le\frac\varepsilon4\cdot2+\frac\varepsilon2=\varepsilon .
\]
Donc $d_X(f_j,f)\to0$.

\medskip
\noindent\textbf{(c)} \emph{Étape 1 : la convergence pour $d_X$ est intrinsèque.} Montrons que
\[
d_X(f_j,f)\to0\iff f_j\to f\ \text{uniformément sur tout compact } K\subset O .
\]
Le sens $(\Leftarrow)$ vient de (b) puisque les $K_n$ sont des compacts de $O$. Pour $(\Rightarrow)$ : si $K\subset O$ est compact, il est contenu dans un $K_{n_0}$ (complément de l'exercice officiel~3), donc $\norm{f_j-f}_K\le\norm{f_j-f}_{K_{n_0}}\to0$ par (b). Le membre de droite ne fait plus référence à la suite $(K_n)$.

\emph{Étape 2 : deux distances ayant les mêmes suites convergentes ont la même topologie.} Soit $d'_X$ la distance construite avec une autre suite exhaustive $(K'_n)$. Par l'étape 1, une suite converge vers $f$ pour $d_X$ si et seulement si elle converge vers $f$ pour $d'_X$. Par la caractérisation séquentielle des fermés (Proposition~1.3), $d_X$ et $d'_X$ ont les mêmes fermés, donc (par complémentaire) les mêmes ouverts : elles définissent la même topologie sur $\Cz(O)$.
\end{solution}

\begin{pointcle}
\textbf{Fabriquer une distance à partir d'une famille dénombrable de semi-normes} $(p_n)_n$ : $d=\sum_n2^{-n}\min\{1,p_n(f-g)\}$. Le facteur $2^{-n}$ assure la convergence, le $\min\{1,\cdot\}$ borne chaque terme, la sous-additivité de $\min\{1,\cdot\}$ donne l'inégalité triangulaire, et la séparation vient de ce que la famille $(p_n)$ est \emph{séparante} ($p_n(f)=0\ \forall n\Rightarrow f=0$). La convergence pour $d$ est la convergence pour \emph{chaque} $p_n$, ce qui se prouve par l'argument « couper la queue de la série » de (b). Cet espace $(\Cz(O),d_X)$ est complet (exercice : passer par la complétude de chaque $\Cz(K_n)$) ; on parle d'\emph{espace de Fréchet}.
\end{pointcle}

\begin{piege}
\textbf{Ce n'est pas une distance issue d'une norme} (Exercice~2.3 du poly). Idée : $d_X$ n'est pas homogène ($d_X(0,\lambda f)\ne|\lambda|d_X(0,f)$ dès que les minima saturent). Plus profondément, pour $N$ fixé, le sous-espace $V_N:=\{f : f=0\text{ sur }K_N\}$ est \emph{contenu dans toute boule $B_{d_X}(0,\rho)$} dès que $2^{-N}<\rho$ (car alors $d_X(0,f)\le\sum_{n>N}2^{-n}<\rho$ pour $f\in V_N$, les termes $n\le N$ étant nuls). Si $d_X$ provenait d'une norme $\mathcal N$, la boule $B_{\mathcal N}(0,1)$ contiendrait une boule $B_{d_X}(0,\rho)\supset V_N$, donc le sous-espace $V_N$ tout entier, ce qui force $\mathcal N=0$ sur $V_N$ ; or $V_N\ne\{0\}$ ($O\ne K_N$ car un ouvert non vide de $\R^d$ n'est jamais compact, et on construit une fonction continue non nulle nulle sur $K_N$ via $d(K_N,\cdot)$).

\textbf{Sur l'énoncé :} la feuille fait démarrer la somme à $n\ge2$ ; cela ne change rien (les $K_n$, $n\ge2$, recouvrent encore $O$ par croissance, et $K_0,K_1\subset K_2$), on a rédigé avec $n\ge0$ comme dans l'Exemple~2.5.
\end{piege}

\begin{exo}{Exercice officiel 5 -- Séparabilité de $\Cz(K;\R)$}
Soit $K$ un compact de $\R^d$. Montrer que $\Cz(K;\R)$ (muni de la norme uniforme) est séparable.
\end{exo}

\begin{solution}
Si $K=\emptyset$ il n'y a rien à dire ; supposons $K\ne\emptyset$. Il s'agit d'exhiber une partie \emph{dénombrable} et \emph{dense} (Définition~1.13).

\emph{Étape 1 : les polynômes sont denses (Stone--Weierstrass).} Notons $\mathcal P$ l'ensemble des fonctions polynomiales en $d$ variables à coefficients réels, restreintes à $K$. C'est une sous-algèbre unitaire de $\Cz(K;\R)$ (stable par somme, produit, contient la constante $1$). Elle \emph{sépare les points} : si $x\ne y$ dans $K$, une coordonnée diffère, $x_i\ne y_i$, et la fonction coordonnée $p_i:z\mapsto z_i$ est dans $\mathcal P$ avec $p_i(x)\ne p_i(y)$. Le Théorème~2.2 s'applique : $\mathcal P$ est dense dans $\Cz(K;\R)$.

\emph{Étape 2 : les polynômes à coefficients rationnels sont dénombrables.} Notons $\mathcal P_\Q\subset\mathcal P$ les polynômes à coefficients dans $\Q$. Un tel polynôme est déterminé par un degré total $m$ et une famille finie de rationnels (ses coefficients, indexés par les multi-indices $\alpha\in\N^d$ avec $|\alpha|\le m$). Pour $m$ fixé il y en a autant que d'éléments de $\Q^{c_m}$ (avec $c_m$ le nombre de multi-indices de longueur $\le m$), ensemble dénombrable ; $\mathcal P_\Q$ est une union dénombrable (sur $m$) d'ensembles dénombrables, donc dénombrable.

\emph{Étape 3 : $\mathcal P_\Q$ est dense dans $\mathcal P$.} Comme $K$ est borné, $M_\alpha:=\sup_{x\in K}|x^\alpha|<+\infty$ pour tout multi-indice $\alpha$. Soit $P=\sum_{|\alpha|\le m}a_\alpha x^\alpha\in\mathcal P$ et $\varepsilon>0$. Par densité de $\Q$ dans $\R$, choisissons $q_\alpha\in\Q$ avec $|a_\alpha-q_\alpha|\,M_\alpha\le\varepsilon/c_m$, et posons $Q:=\sum q_\alpha x^\alpha\in\mathcal P_\Q$. Alors
\[
\norm{P-Q}_K\le\sum_{|\alpha|\le m}|a_\alpha-q_\alpha|\,M_\alpha\le\varepsilon .
\]

\emph{Étape 4 : la densité est transitive.} Soit $f\in\Cz(K;\R)$ et $\varepsilon>0$. Par l'étape 1, il existe $P\in\mathcal P$ avec $\norm{f-P}_K\le\varepsilon/2$ ; par l'étape 3, $Q\in\mathcal P_\Q$ avec $\norm{P-Q}_K\le\varepsilon/2$. Donc $\norm{f-Q}_K\le\varepsilon$ : $\mathcal P_\Q$ est dense dans $\Cz(K;\R)$, et dénombrable. L'espace est séparable.
\end{solution}

\begin{methode}
\textbf{Prouver une séparabilité} suit presque toujours ce chemin : (1) un théorème de densité fournit une partie dense « à paramètres réels » (polynômes, fonctions en escalier, suites à support fini) ; (2) on rend les paramètres rationnels, ce qui donne une partie dénombrable ; (3) on vérifie que l'approximation par des paramètres rationnels ne coûte qu'un $\varepsilon$ et l'on conclut par transitivité de la densité. Comparer avec la Proposition~2.8 (séparabilité de $L^p$) et le Corollaire~2.2 ($\ell^p$).
\end{methode}

\begin{piege}
\textbf{On ne peut pas appliquer Stone--Weierstrass directement à $\mathcal P_\Q$.} Le Théorème~2.2 demande une \emph{sous-algèbre} de $\Cz(K;\R)$, donc en particulier un sous-espace vectoriel \emph{réel}. Or $\mathcal P_\Q$ n'est stable que par multiplication par des rationnels ($\sqrt2\cdot\mathbf 1\notin\mathcal P_\Q$) : c'est une $\Q$-algèbre, et l'hypothèse « $\R$-espace vectoriel » est réellement utilisée dans la preuve (interpolation $v_1+(v_2-v_1)\frac{g-g(a)}{g(b)-g(a)}$ avec $v_1,v_2$ réels, approximation de $|f|$ par des polynômes à coefficients réels). L'étape 3 ci-dessus est précisément ce qui remplace cette hypothèse manquante : elle montre à la main que $\adh{\mathcal P_\Q}\supset\mathcal P$.
\end{piege}

\bigskip
\noindent\textbf{Seconde solution, sans Stone--Weierstrass : par partitions de l'unité.} Elle suit le schéma de la preuve de la Proposition~2.8 (partition fine, uniforme continuité, valeurs rationnelles), en remplaçant les indicatrices de cellules, qui ne sont pas continues, par des « fonctions chapeau » continues. Elle a l'avantage de fonctionner pour tout espace métrique compact $X$ à la place de $K\subset\R^d$.

\begin{solution}[bis]
\textbf{Lemme (partition de l'unité continue subordonnée à un recouvrement fini).} \emph{Soit $(X,d)$ un espace métrique et $U_1,\dots,U_p$ des ouverts recouvrant $X$. Il existe des fonctions continues $\varphi_1,\dots,\varphi_p:X\to[0,1]$ telles que}
\[
\sum_{i=1}^p\varphi_i\equiv1\qquad\text{et}\qquad\varphi_i(x)\ne0\ \Longrightarrow\ x\in U_i\quad(1\le i\le p).
\]

\emph{Preuve du lemme.} Posons $\psi_i(x):=d(X\setminus U_i,\,x)$, avec la convention $\psi_i\equiv1$ si $U_i=X$. Chaque $\psi_i$ est continue ($1$-lipschitzienne, TD1 off.~6), positive, et $\psi_i(x)>0\iff x\notin\adh{X\setminus U_i}=X\setminus U_i\iff x\in U_i$ (Corollaire~1.1, $X\setminus U_i$ étant fermé). Comme les $U_i$ recouvrent $X$, pour tout $x$ l'un au moins des $\psi_i(x)$ est $>0$ : la somme $\Psi:=\sum_j\psi_j$ est continue et \emph{ne s'annule pas}. On pose $\varphi_i:=\psi_i/\Psi$. Alors $\varphi_i$ est continue (quotient de fonctions continues à dénominateur non nul), à valeurs dans $[0,1]$, $\sum_i\varphi_i=\Psi/\Psi=1$, et $\varphi_i(x)\ne0\iff\psi_i(x)\ne0\iff x\in U_i$. \hfill$\square$

\emph{Dans le cas concret utilisé ci-dessous}, $U_i=B(x_i,2\varepsilon)$ et l'on peut prendre directement $\psi_i(x)=\max\big(0,\,2\varepsilon-d(x,x_i)\big)$ : c'est un « chapeau » de hauteur $2\varepsilon$ centré en $x_i$, nul hors de $B(x_i,2\varepsilon)$ ; la normalisation par $\Psi$ est ce qui rend l'argument indépendant de toute régularité du réseau. (Sur une grille régulière de $\R^d$ de pas $h$, les produits de fonctions tente $\prod_k\max(0,1-|x_k-(x_i)_k|/h)$ forment déjà une partition de l'unité sans normalisation ; c'est l'interpolation $Q_1$ des éléments finis.)

\medskip
\noindent\emph{Étape 1 : des réseaux finis.} Soit $X$ un espace métrique compact non vide (ici $X=K$). Pour $n\ge1$, la précompacité de $X$ (Proposition~1.13) fournit un $2^{-n}$-réseau fini $R_n=\{x^n_1,\dots,x^n_{p_n}\}$, \ie{} $X=\bigcup_iB(x^n_i,2^{-n})$. On applique le lemme au recouvrement ouvert $U_i:=B(x^n_i,2^{1-n})$ : on obtient $\varphi^n_1,\dots,\varphi^n_{p_n}$ continues, positives, de somme $1$, avec $\varphi^n_i(x)\ne0\Rightarrow d(x,x^n_i)<2^{1-n}$.

\emph{Étape 2 : l'opérateur d'interpolation.} Pour $f\in\Cz(X;\R)$, posons
\[
I_nf:=\sum_{i=1}^{p_n}f(x^n_i)\,\varphi^n_i\ \in\Cz(X;\R).
\]
Soit $\omega_f(\delta):=\sup\{|f(x)-f(y)| : d(x,y)<\delta\}$ le module de continuité de $f$ ; par le théorème de Heine (Théorème~1.3), $\omega_f(\delta)\to0$ quand $\delta\to0$. Pour $x\in X$, en utilisant $\sum_i\varphi^n_i(x)=1$,
\begin{align*}
\big|f(x)-I_nf(x)\big|&=\Big|\sum_{i=1}^{p_n}\big(f(x)-f(x^n_i)\big)\varphi^n_i(x)\Big|
\le\sum_{i\,:\,\varphi^n_i(x)\ne0}\big|f(x)-f(x^n_i)\big|\,\varphi^n_i(x)\\
&\le\omega_f(2^{1-n})\sum_i\varphi^n_i(x)=\omega_f(2^{1-n}),
\end{align*}
car seuls contribuent les indices $i$ tels que $\varphi^n_i(x)\ne0$, pour lesquels $d(x,x^n_i)<2^{1-n}$. Donc $\norm{f-I_nf}_\infty\le\omega_f(2^{1-n})\to0$.

\emph{Étape 3 : rationalisation et dénombrabilité.} Posons
\[
\mathcal D:=\Big\{\sum_{i=1}^{p_n}r_i\,\varphi^n_i\ :\ n\ge1,\ (r_1,\dots,r_{p_n})\in\Q^{p_n}\Big\}.
\]
Pour $n$ fixé, l'ensemble des combinaisons est en bijection avec $\Q^{p_n}$, dénombrable ; $\mathcal D$ est une union dénombrable (sur $n$) d'ensembles dénombrables, donc dénombrable. Il est dense : soit $f\in\Cz(X;\R)$ et $\varepsilon>0$ ; on choisit $n$ avec $\omega_f(2^{1-n})\le\varepsilon/2$, puis des rationnels $r_i$ avec $|r_i-f(x^n_i)|\le\varepsilon/2$, et $g:=\sum_ir_i\varphi^n_i\in\mathcal D$ vérifie
\[
\norm{f-g}_\infty\le\norm{f-I_nf}_\infty+\Big\|\sum_i\big(f(x^n_i)-r_i\big)\varphi^n_i\Big\|_\infty\le\frac\varepsilon2+\frac\varepsilon2\sup_x\sum_i\varphi^n_i(x)=\varepsilon .
\]
Ainsi $\Cz(X;\R)$ est séparable, pour tout espace métrique compact $X$.
\end{solution}

\begin{pointcle}
\textbf{Pourquoi des chapeaux et non des indicatrices.} La preuve de la Proposition~2.8 approche $f$ par une fonction constante par cellules ; en norme $L^p$ c'est licite, mais pour $\norm\cdot_\infty$ l'approximant doit lui-même être continu, sinon il n'est pas dans l'espace. La partition de l'unité fournit exactement des « indicatrices continues » : positives, de somme $1$, localisées. Les deux ingrédients de fond sont les mêmes : \emph{compacité} (réseau fini, uniforme continuité) et \emph{rationalisation} des valeurs. Le lemme lui-même est le prolongement naturel du « lemme d'Urysohn métrique » (TD1 off.~6, commentaire) et servira à nouveau pour les distributions (chapitre~6, recollement local).
\end{pointcle}

\begin{commentaire}
\begin{itemize}
  \item Le même argument, avec $\Q+i\Q$ et le Théorème~2.3, donne la séparabilité de $\Cz(K;\C)$ ; pour $\Cz(\Sun;\C)$ on peut aussi utiliser les polynômes trigonométriques à coefficients dans $\Q+i\Q$ (Exemple~2.4).
  \item Plus généralement $\Cz(X;\R)$ est séparable pour tout espace métrique compact $X$ : $X$ est séparable (précompact, Proposition~1.13, donc réunion dénombrable de boules finies), et si $(x_n)$ est dense, l'algèbre engendrée par $1$ et les fonctions $d(x_n,\cdot)$ sépare les points ; on conclut par Stone--Weierstrass et coefficients rationnels.
  \item L'exercice officiel~9 montrera que ce n'est pas le cas de tous les espaces de fonctions « naturels » : $\ell^\infty$ et $L^\infty$ ne sont pas séparables.
\end{itemize}
\end{commentaire}

\begin{exo}{Exercice officiel 6 -- Trois expressions de la norme d'opérateur}
Soient $(E,\norm\cdot_E)$ et $(F,\norm\cdot_F)$ deux EVN et $\norm T_{\Lc(E,F)}:=\sup_{\norm x_E\le1}\norm{T(x)}_F$. Montrer que pour tout $T\in\Lc(E,F)$,
\[
\norm T_{\Lc(E,F)}=\sup_{\norm x_E=1}\norm{T(x)}_F=\sup_{x\in E\setminus\{0\}}\frac{\norm{T(x)}_F}{\norm x_E}.
\]
\end{exo}

\begin{solution}
On suppose $E\ne\{0\}$ (sinon les deux derniers supremums portent sur l'ensemble vide et l'énoncé est sans objet ; le premier vaut $0$). Notons
\[
A:=\sup_{\norm x_E\le1}\norm{Tx}_F,\qquad B:=\sup_{\norm x_E=1}\norm{Tx}_F,\qquad C:=\sup_{x\ne0}\frac{\norm{Tx}_F}{\norm x_E}.
\]
Ces trois quantités sont finies car $T$ est continue (Théorème~2.4 (iii)-(iv)).

\emph{$B\le A$.} La sphère $\{\norm x_E=1\}$ est contenue dans la boule $\{\norm x_E\le1\}$ : un sup sur un ensemble plus petit est plus petit.

\emph{$B=C$.} Pour $x\ne0$, posons $u:=x/\norm x_E$, de norme $1$. Par linéarité de $T$ et homogénéité de la norme,
\[
\frac{\norm{Tx}_F}{\norm x_E}=\norm{T\Big(\frac{x}{\norm x_E}\Big)}_F=\norm{Tu}_F\le B ,
\]
d'où $C\le B$ en passant au sup. Réciproquement, pour $\norm x_E=1$, $\norm{Tx}_F=\norm{Tx}_F/\norm x_E\le C$, d'où $B\le C$.

\emph{$A\le C$.} Soit $\norm x_E\le1$. Si $x=0$, $\norm{Tx}_F=0\le C$. Sinon,
\[
\norm{Tx}_F=\norm x_E\cdot\frac{\norm{Tx}_F}{\norm x_E}\le\norm x_E\cdot C\le C .
\]
Passant au sup, $A\le C$.

\emph{Conclusion.} $B\le A\le C=B$, donc $A=B=C$.
\end{solution}

\begin{pointcle}
La troisième expression dit que $\norm T_{\Lc(E,F)}$ est \emph{la meilleure constante} $M$ dans l'inégalité $\norm{Tx}_F\le M\norm x_E$ (Remarque~2.7) : c'est ainsi qu'on l'utilise en pratique, et c'est ainsi qu'on la calcule (exercice officiel~7) : \textbf{une majoration} $\norm{Tx}\le M\norm x$ pour tout $x$ donne $\opnorm T\le M$, puis \textbf{un vecteur test} $x_0$ (ou une suite de vecteurs tests) avec $\norm{Tx_0}$ proche de $M\norm{x_0}$ donne $\opnorm T\ge M$. Tout repose sur l'\emph{homogénéité} : normaliser un vecteur ne change pas le rapport $\norm{Tx}/\norm x$.
\end{pointcle}

\begin{commentaire}
On a même $\opnorm T=\sup_{\norm x_E<1}\norm{Tx}_F$ (boule ouverte) : pour $\norm x_E=1$, $\norm{T(tx)}_F=t\norm{Tx}_F\to\norm{Tx}_F$ quand $t\to1^-$. En revanche, le sup \emph{n'est pas toujours atteint} en dimension infinie : sur $c_0(\N)$ (suites tendant vers $0$, norme $\sup$), la forme linéaire $T\mathbf u:=\sum_{n\ge1}2^{-n}u_n$ vérifie $|T\mathbf u|\le\norm{\mathbf u}_\infty$ avec égalité seulement si $u_n=\pm1$ pour tout $n$, impossible dans $c_0$ ; pourtant $\opnorm T=1$ (tester avec $\mathbf u=(1,\dots,1,0,0,\dots)$). L'exercice officiel~7 (b.2) présente le même phénomène.
\end{commentaire}

\begin{exo}{Exercice officiel 7 (Exercice 2.5 du poly) -- Coefficients de Fourier et noyau de Dirichlet}
On considère, pour $f\in\Cz(\Sun;\C)$ (identifié aux fonctions continues et $2\pi$-périodiques sur $\R$, Exemple~2.2) et $n\in\Z$,
\[
c_n(f):=\frac1{2\pi}\int_0^{2\pi}f\,\overline{e_n},\qquad e_n:t\mapsto e^{int}.
\]
\begin{enumerate}[label=(\alph*)]
  \item Montrer que $f\mapsto c_n(f)$ est une forme linéaire continue sur $\Cz(\Sun;\C)$ et calculer sa norme.
  \item Pour $N\in\N$, on cherche la norme de $\Phi_N:f\mapsto\sum_{|n|\le N}c_n(f)$.
  \begin{enumerate}[label=(b.\arabic*)]
    \item Montrer que $\opnorm{\Phi_N}\le(2\pi)^{-1}\int_0^{2\pi}|D_N|$, où $D_N:=\sum_{|n|\le N}e_n$ (noyau de Dirichlet).
    \item En considérant $f_n:=D_N\,(n^{-1}+|D_N|)^{-1}$, transformer l'inégalité précédente en égalité.
  \end{enumerate}
\end{enumerate}
\end{exo}

\begin{solution}
\noindent\textbf{(a)} \emph{Linéarité :} elle découle de la linéarité de l'intégrale. \emph{Continuité et majoration de la norme :} comme $|\overline{e_n}(t)|=1$,
\[
|c_n(f)|\le\frac1{2\pi}\int_0^{2\pi}|f(t)|\,|\overline{e_n}(t)|\,dt\le\frac1{2\pi}\int_0^{2\pi}\norm f_\infty\,dt=\norm f_\infty .
\]
Par le Théorème~2.4 (iv), $c_n$ est continue et $\opnorm{c_n}\le1$ (exercice officiel~6). \emph{Vecteur test :} $f=e_n$ vérifie $\norm{e_n}_\infty=1$ et
\[
c_n(e_n)=\frac1{2\pi}\int_0^{2\pi}e^{int}e^{-int}\,dt=1 ,
\]
donc $\opnorm{c_n}\ge|c_n(e_n)|/\norm{e_n}_\infty=1$. Finalement $\opnorm{c_n}=1$.

\medskip
\noindent\textbf{(b.1)} Par linéarité de l'intégrale,
\[
\Phi_N(f)=\sum_{|n|\le N}\frac1{2\pi}\int_0^{2\pi}f\,\overline{e_n}=\frac1{2\pi}\int_0^{2\pi}f\,\overline{D_N},
\qquad\overline{D_N}=\sum_{|n|\le N}\overline{e_n}=\sum_{|n|\le N}e_{-n}=D_N .
\]
Le noyau $D_N$ est donc \emph{à valeurs réelles} : $D_N(t)=1+2\sum_{n=1}^N\cos(nt)$. Ainsi
\[
|\Phi_N(f)|\le\frac1{2\pi}\int_0^{2\pi}|f|\,|D_N|\le\norm f_\infty\cdot\frac1{2\pi}\int_0^{2\pi}|D_N| ,
\]
ce qui montre que $\Phi_N$ est continue et $\opnorm{\Phi_N}\le L_N:=\dfrac1{2\pi}\displaystyle\int_0^{2\pi}|D_N|$.

\medskip
\noindent\textbf{(b.2)} Pour $n\ge1$, posons $f_n:=\dfrac{D_N}{\frac1n+|D_N|}$. Le dénominateur est $\ge\frac1n>0$ et continu, donc $f_n$ est continue et $2\pi$-périodique : $f_n\in\Cz(\Sun;\C)$ (à valeurs réelles). De plus
\[
|f_n|=\frac{|D_N|}{\frac1n+|D_N|}<1,\qquad\text{donc}\qquad\norm{f_n}_\infty\le1 .
\]
Calculons $\Phi_N(f_n)$, en utilisant $D_N\cdot D_N=|D_N|^2$ ($D_N$ réel) et la minoration $\dfrac{|D_N|^2}{\frac1n+|D_N|}=|D_N|-\dfrac{\frac1n|D_N|}{\frac1n+|D_N|}\ge|D_N|-\dfrac1n$ :
\[
\Phi_N(f_n)=\frac1{2\pi}\int_0^{2\pi}\frac{|D_N|^2}{\frac1n+|D_N|}\ \ge\ \frac1{2\pi}\int_0^{2\pi}\Big(|D_N|-\frac1n\Big)=L_N-\frac1n .
\]
Par l'exercice officiel~6, $\opnorm{\Phi_N}\ge\dfrac{|\Phi_N(f_n)|}{\norm{f_n}_\infty}\ge L_N-\dfrac1n$ pour tout $n\ge1$. En faisant $n\to\infty$ : $\opnorm{\Phi_N}\ge L_N$, et avec (b.1),
\[
\opnorm{\Phi_N}=L_N=\frac1{2\pi}\int_0^{2\pi}|D_N(t)|\,dt .
\]
\end{solution}

\begin{pointcle}
\textbf{Calculer une norme d'opérateur} : majoration « brutale » via l'inégalité triangulaire intégrale, puis vecteur test. Ici le vecteur test idéal serait $\sgn(D_N)$ (pour avoir $f\,D_N=|D_N|$ avec $|f|=1$), mais il est \emph{discontinu} là où $D_N$ change de signe. On le \emph{régularise} en $f_n=D_N/(\frac1n+|D_N|)$, qui est continue, de module $<1$, et converge vers $\sgn(D_N)$ ; le sup n'est pas atteint mais est approché. Ce procédé (approcher un « signe » par des fonctions continues) revient constamment, par exemple pour la représentation duale des normes $L^p$ (Lemme~2.2 du poly), premier pas vers l'identification du dual de $L^p$ (Théorème~5.6).
\end{pointcle}

\begin{commentaire}
\textbf{Pourquoi cet exercice ?} $\Phi_N(f)$ est la $N$-ième somme partielle de Fourier de $f$ évaluée en $0$ : $S_Nf(0)=\sum_{|n|\le N}c_n(f)$. Les constantes $L_N$ sont les \emph{constantes de Lebesgue}. Avec la forme close $D_N(t)=\dfrac{\sin((N+\frac12)t)}{\sin(t/2)}$, on montre que $L_N\ge c\ln N\to+\infty$. Le théorème de Banach--Steinhaus (Théorème~3.7, chapitre~3 ; voir les Exemples~3.4--3.5 développés dans le TD0) transformera cette explosion des normes en un résultat frappant : \emph{il existe des fonctions continues dont la série de Fourier diverge en $0$}. Cet exercice est donc la première étape d'un contre-exemple célèbre (du Bois-Reymond, 1873).
\end{commentaire}

\begin{exo}{Exercice officiel 8 (Exercice 2.7 du poly) -- Lemme de Riemann--Lebesgue}
\begin{enumerate}[label=(\alph*)]
  \item Montrer que l'ensemble $c_0(\Z)$ des suites tendant vers $0$ en $\pm\infty$ est un sous-espace fermé de $\ell^\infty(\Z)$.
  \item Montrer que $\mathcal F:f\mapsto(c_n(f))_{n\in\Z}$ envoie continûment $\Cz(\Sun;\C)$ dans $\ell^\infty(\Z)$.
  \item En utilisant la densité des fonctions en escalier dans $\Cz(\Sun;\C)$, montrer que l'image de $\mathcal F$ est contenue dans $c_0(\Z)$.
\end{enumerate}
\end{exo}

\begin{solution}
\noindent\textbf{(a)} \emph{Sous-espace.} Une suite tendant vers $0$ en $\pm\infty$ est bornée, donc $c_0(\Z)\subset\ell^\infty(\Z)$ ; la suite nulle y est ; si $\mathbf u,\mathbf v\in c_0(\Z)$ et $\lambda\in\K$, $|u_n+\lambda v_n|\le|u_n|+|\lambda||v_n|\to0$ quand $|n|\to\infty$.

\emph{Fermé.} Utilisons la caractérisation séquentielle. Soit $(\mathbf u^{(k)})_k$ une suite d'éléments de $c_0(\Z)$ convergeant dans $\ell^\infty(\Z)$ vers $\mathbf u$, \ie{} $\norm{\mathbf u^{(k)}-\mathbf u}_\infty\to0$. Soit $\varepsilon>0$. Choisissons $k$ tel que $\norm{\mathbf u^{(k)}-\mathbf u}_\infty\le\varepsilon/2$, puis, comme $\mathbf u^{(k)}\in c_0(\Z)$, un entier $M$ tel que $|u^{(k)}_n|\le\varepsilon/2$ pour $|n|\ge M$. Alors, pour $|n|\ge M$,
\[
|u_n|\le|u_n-u^{(k)}_n|+|u^{(k)}_n|\le\norm{\mathbf u-\mathbf u^{(k)}}_\infty+\frac\varepsilon2\le\varepsilon .
\]
Donc $u_n\to0$ quand $|n|\to\infty$ : $\mathbf u\in c_0(\Z)$.

\medskip
\noindent\textbf{(b)} Par l'exercice officiel~7 (a), $|c_n(f)|\le\norm f_\infty$ pour tout $n\in\Z$ : la suite $\mathcal F(f)=(c_n(f))_n$ est bornée, donc $\mathcal F(f)\in\ell^\infty(\Z)$, avec
\[
\norm{\mathcal F(f)}_\infty=\sup_{n\in\Z}|c_n(f)|\le\norm f_\infty .
\]
L'application $\mathcal F$ est linéaire (chaque $c_n$ l'est), et cette inégalité est le critère (iv) du Théorème~2.4 : $\mathcal F$ est continue, de norme $\le1$ (en fait $=1$ : tester $f=e_0=1$, pour laquelle $c_0(f)=1$).

\medskip
\noindent\textbf{(c)} \emph{Étape 1 : extension de $\mathcal F$ aux fonctions en escalier, avec la même constante.} La formule $c_n(g)=\frac1{2\pi}\int_0^{2\pi}g\,\overline{e_n}$ garde un sens pour toute fonction $g$ bornée et intégrable sur $[0,2\pi]$, avec toujours $|c_n(g)|\le\norm g_\infty$. Notons $\mathcal E$ l'espace vectoriel engendré par $\Cz(\Sun;\C)$ et les fonctions en escalier, muni de la norme uniforme. L'application $\mathcal F:g\mapsto(c_n(g))_{n\in\Z}$ est linéaire de $\mathcal E$ dans $\ell^\infty(\Z)$ et
\begin{equation}\label{eq:lip}
\norm{\mathcal F(g)-\mathcal F(h)}_\infty=\sup_n|c_n(g-h)|\le\norm{g-h}_\infty\qquad(g,h\in\mathcal E):
\end{equation}
$\mathcal F$ est $1$-lipschitzienne, donc continue, sur $\mathcal E$. C'est la question (b), étendue au-delà des fonctions continues ; sans cette extension, la continuité de (b) ne dirait rien des fonctions en escalier.

\emph{Étape 2 : les fonctions en escalier sont envoyées dans $c_0(\Z)$.} Pour $0\le a<b\le2\pi$ et $n\ne0$,
\[
c_n(\mathbf 1_{[a,b[})=\frac1{2\pi}\int_a^be^{-int}\,dt=\frac1{2\pi}\cdot\frac{e^{-inb}-e^{-ina}}{-in},
\qquad\text{donc}\qquad
|c_n(\mathbf 1_{[a,b[})|\le\frac1{2\pi}\cdot\frac2{|n|}\xrightarrow[|n|\to\infty]{}0 .
\]
Par linéarité, $\mathcal F(g)\in c_0(\Z)$ pour toute fonction en escalier $g$ (combinaison linéaire finie d'indicatrices d'intervalles).

\emph{Étape 3 : densité.} Soit $f\in\Cz(\Sun;\C)$ ; $f$ est continue sur le compact $[0,2\pi]$, donc uniformément continue (Théorème~1.3 de Heine). Pour $m\ge1$, posons $t_k:=2\pi k/m$ et $g_m:=\sum_{k=0}^{m-1}f(t_k)\mathbf 1_{[t_k,t_{k+1}[}$. Pour $t\in[t_k,t_{k+1}[$, $|f(t)-g_m(t)|=|f(t)-f(t_k)|\le\omega_f(2\pi/m)$, où $\omega_f$ est le module de continuité uniforme de $f$ : donc $\norm{f-g_m}_\infty\to0$.

\emph{Étape 4 : conclusion par « dense $+$ continu $+$ fermé ».} Par l'étape 1, $\mathcal F(g_m)\to\mathcal F(f)$ dans $\ell^\infty(\Z)$ ; par l'étape 2, $\mathcal F(g_m)\in c_0(\Z)$ pour tout $m$ ; par la question (a), $c_0(\Z)$ est fermé dans $\ell^\infty(\Z)$ : la limite $\mathcal F(f)$ appartient donc à $c_0(\Z)$ (caractérisation séquentielle des fermés, Proposition~1.3).

\emph{Remarque.} On peut aussi conclure « à la main » sans citer (a) : pour $\varepsilon>0$, choisir $m$ avec $\norm{f-g_m}_\infty\le\varepsilon/2$, puis $M$ avec $|c_n(g_m)|\le\varepsilon/2$ pour $|n|\ge M$, et écrire $|c_n(f)|\le|c_n(f-g_m)|+|c_n(g_m)|\le\varepsilon$. Mais cet argument $\varepsilon/2$ n'est rien d'autre que la preuve de (a) recopiée avec $\mathbf u^{(k)}=\mathcal F(g_k)$ : l'énoncé a justement posé (a) et (b) pour qu'on n'ait pas à le refaire.
\end{solution}

\begin{pointcle}
\textbf{Le principe « dense $+$ continu $+$ fermé ».} Si une application continue $\mathcal F$ envoie une partie dense $D$ dans un fermé $C$, alors elle envoie tout l'espace dans $C$ : $\mathcal F(\adh D)\subset\adh{\mathcal F(D)}\subset\adh C=C$. C'est exactement la structure de (c), et c'est pour cela que (a) demandait que $c_0(\Z)$ soit \emph{fermé} et (b) que $\mathcal F$ soit \emph{continue}. Le résultat est le \emph{lemme de Riemann--Lebesgue} : les coefficients de Fourier d'une fonction continue (et même $L^1$, par le même argument de densité) tendent vers $0$.
\end{pointcle}

\begin{commentaire}
Deux variantes de la preuve de (c), instructives :
\begin{itemize}
  \item \emph{Sans fonctions en escalier :} le changement de variable $t\mapsto t+\pi/n$ ($n\ne0$) et la périodicité donnent $c_n(f)=-\frac1{2\pi}\int_0^{2\pi}f(t+\frac\pi n)e^{-int}dt$, donc $2|c_n(f)|\le\frac1{2\pi}\int_0^{2\pi}|f(t)-f(t+\frac\pi n)|dt\le\omega_f(\pi/|n|)\to0$, où $\omega_f$ est le module de continuité uniforme de $f$.
  \item \emph{Avec Stone--Weierstrass :} les polynômes trigonométriques sont denses (Exemple~2.4) et leurs coefficients de Fourier sont nuls au-delà d'un rang ; on conclut par le même $\varepsilon/2$.
\end{itemize}
Dans les deux cas, ce qui décroît est la « régularité » de $f$ : plus $f$ est régulière, plus $c_n(f)$ décroît vite (une intégration par parties donne $|c_n(f)|\le\norm{f'}_\infty/|n|$ pour $f\in\mathcal C^1$).
\end{commentaire}

\begin{exo}{Exercice officiel 9 (Exercices 2.6 et 2.8 du poly) -- Non-séparabilité de $\ell^\infty$ et $L^\infty$}
\begin{enumerate}[label=(\alph*)]
  \item Montrer que l'ensemble des suites ne prenant comme valeurs que $0$ et $1$ est indénombrable. En déduire que $\ell^\infty(\N)$ n'est pas séparable.
  \item Soit $\Omega$ un ouvert non vide de $\R^d$. Montrer que $L^\infty(\Omega)$ n'est pas séparable.
\end{enumerate}
\end{exo}

\begin{solution}
\noindent\textbf{(a)} \emph{$\{0,1\}^\N$ est indénombrable (argument diagonal de Cantor).} Supposons par l'absurde qu'on puisse énumérer ses éléments : $\{0,1\}^\N=\{\mathbf u^{(k)}:k\in\N\}$. Définissons la suite $\mathbf v$ par $v_n:=1-u^{(n)}_n\in\{0,1\}$. Pour tout $k$, $v_k\ne u^{(k)}_k$, donc $\mathbf v\ne\mathbf u^{(k)}$ : $\mathbf v$ n'est pas dans l'énumération, contradiction.

\emph{Des boules disjointes en quantité indénombrable.} Pour $\mathbf u\ne\mathbf v$ dans $\{0,1\}^\N\subset\ell^\infty(\N)$, il existe un indice $n$ avec $|u_n-v_n|=1$, et tous les écarts sont $\le1$ : $\norm{\mathbf u-\mathbf v}_\infty=1$. Les boules ouvertes $B(\mathbf u,\frac12)$, $\mathbf u\in\{0,1\}^\N$, sont donc \emph{deux à deux disjointes} : si $\mathbf w$ appartenait à $B(\mathbf u,\frac12)\cap B(\mathbf v,\frac12)$, on aurait $\norm{\mathbf u-\mathbf v}_\infty\le\norm{\mathbf u-\mathbf w}_\infty+\norm{\mathbf w-\mathbf v}_\infty<1$.

\emph{Non-séparabilité.} Supposons qu'il existe une partie $D\subset\ell^\infty(\N)$ dénombrable et dense. Par la caractérisation (iii) de la Proposition~1.4, $D$ rencontre tout ouvert non vide, en particulier chaque boule $B(\mathbf u,\frac12)$ : choisissons $\psi(\mathbf u)\in D\cap B(\mathbf u,\frac12)$. Les boules étant disjointes, $\psi:\{0,1\}^\N\to D$ est \emph{injective}. Mais une injection d'un ensemble indénombrable dans un ensemble dénombrable n'existe pas : contradiction. Donc $\ell^\infty(\N)$ n'est pas séparable.

\medskip
\noindent\textbf{(b)} Rappelons que $\norm f_{L^\infty(\Omega)}$ est le \emph{supremum essentiel} : le plus petit $M$ tel que $|f|\le M$ presque partout. En particulier, pour un ensemble mesurable $A$, $\norm{\mathbf 1_A}_{L^\infty}=1$ si $\lambda_d(A)>0$ et $0$ sinon.

\emph{Une famille indénombrable à distances mutuelles $1$.} Comme $\Omega$ est un ouvert non vide, il contient une boule $B(x_0,r)$ avec $r>0$. Pour $t\in\,]0,r]$, posons $f_t:=\mathbf 1_{B(x_0,t)}\in L^\infty(\Omega)$. Pour $0<s<t\le r$,
\[
f_t-f_s=\mathbf 1_{\{x\,:\,s\le\norm{x-x_0}<t\}} .
\]
La couronne $\{s\le\norm{x-x_0}<t\}$ contient l'ouvert non vide $\{s<\norm{x-x_0}<t\}$, donc est de mesure de Lebesgue strictement positive ; par conséquent $\norm{f_t-f_s}_{L^\infty}=1$.

\emph{Conclusion, comme en (a).} Les boules $B(f_t,\frac12)$, $t\in\,]0,r]$, sont deux à deux disjointes et en quantité indénombrable (l'intervalle $]0,r]$ l'est). Une partie dense devrait rencontrer chacune d'elles, ce qu'une partie dénombrable ne peut faire. Donc $L^\infty(\Omega)$ n'est pas séparable.
\end{solution}

\begin{pointcle}
\textbf{Critère de non-séparabilité :} \emph{si un espace métrique contient une famille indénombrable de points deux à deux à distance $\ge\delta>0$, il n'est pas séparable}. Réciproquement, dans un espace séparable toute famille d'ouverts non vides deux à deux disjoints est dénombrable. Ce critère est l'outil standard ; il montre aussi que $\Cz_b(\R)$ (fonctions continues bornées, norme uniforme) ou $\ell^\infty(\Z)$ ne sont pas séparables.
\end{pointcle}

\begin{piege}
\begin{itemize}
  \item Dans $L^\infty$, « $f_t\ne f_s$ » ne suffit pas : il faut que $f_t-f_s$ soit non nulle \emph{sur un ensemble de mesure strictement positive}, sinon $\norm{f_t-f_s}_{L^\infty}=0$. C'est pour cela qu'on vérifie que la couronne contient un ouvert non vide.
  \item Contraste avec $p<\infty$ : $\ell^p(\N)$ (Corollaire~2.2) et $L^p(\Omega)$ (Proposition~2.8) sont séparables, grâce à la densité des suites à support fini, resp. des fonctions continues à support compact. Dans $\ell^\infty$, $c_{<\infty}(\N)$ n'est pas dense (Proposition~2.3, la suite constante $\mathbf 1$ est à distance $1$ de toute suite à support fini) : c'est précisément ce qui rend possible la non-séparabilité. Le sous-espace fermé $c_0(\N)$, lui, est séparable.
  \item L'indication du poly (« s'inspirer de l'Exercice~2.6 ») suggère une autre construction : des boules disjointes $B_k\subset\Omega$, $k\in\N$, et les fonctions $f_{\mathbf u}:=\sum_ku_k\mathbf 1_{B_k}$ pour $\mathbf u\in\{0,1\}^\N$, qui vérifient encore $\norm{f_{\mathbf u}-f_{\mathbf v}}_{L^\infty}=1$ pour $\mathbf u\ne\mathbf v$.
\end{itemize}
\end{piege}

\begin{exo}{Exercice 7 -- Questions de cours}
\emph{À traiter sans document : énoncé complet, puis démonstration.}
\begin{enumerate}[label=(\alph*)]
  \item Énoncer et démontrer la caractérisation des applications linéaires continues entre EVN (Théorème~2.4).
  \item Énoncer et démontrer l'équivalence des normes sur $\K^d$ (Théorème~2.1).
  \item Énoncer l'inégalité de Young $ab\le\frac{a^p}p+\frac{b^q}q$, puis énoncer et démontrer l'inégalité de Hölder dans $\K^d$ (Lemme~2.1).
  \item Montrer que $c_{<\infty}(\N)$ est dense dans $\ell^p(\N)$ pour $1\le p<\infty$ (Proposition~2.3), et en déduire que $\ell^p(\N)$ est séparable (Corollaire~2.2). Que se passe-t-il pour $p=\infty$ ?
\end{enumerate}
\end{exo}

\begin{solution}
\noindent\textbf{(a)} \emph{Énoncé.} Soient $E,F$ deux EVN et $T:E\to F$ linéaire. Sont équivalentes : (i) $T$ est lipschitzienne ; (ii) $T$ est continue en $0$ ; (iii) $T$ est bornée sur $B_E(0,1)$ ; (iv) il existe $C>0$ tel que $\norm{T(x)}_F\le C\norm x_E$ pour tout $x$.

\emph{Preuve.} (i)$\Rightarrow$(ii) : immédiat. (ii)$\Rightarrow$(iii) : il existe $\delta>0$ tel que $\norm x<\delta$ entraîne $\norm{T(x)}<1$ ; pour $\norm x<1$, $\norm{T(x)}=\delta^{-1}\norm{T(\delta x)}<\delta^{-1}$. (iii)$\Rightarrow$(iv) : soit $M$ une borne de $\norm{T}$ sur $B_E(0,1)$ ; pour $x\ne0$, $\frac x{2\norm x}\in B_E(0,1)$, donc $\norm{T(x)}=2\norm x\,\big\|T\big(\frac x{2\norm x}\big)\big\|\le2M\norm x$. (iv)$\Rightarrow$(i) : par linéarité, $\norm{T(x)-T(y)}=\norm{T(x-y)}\le C\norm{x-y}$.

\medskip
\noindent\textbf{(b)} \emph{Énoncé.} Sur $\K^d$, toutes les normes sont équivalentes.

\emph{Preuve.} Il suffit de comparer une norme $N$ à $\norm\cdot_\infty$. \emph{Majoration :} avec la base canonique $(e_k)$, $N(x)\le\sum_k|x_k|N(e_k)\le\beta\norm x_\infty$, où $\beta:=\sum_kN(e_k)$. En particulier $|N(x)-N(y)|\le N(x-y)\le\beta\norm{x-y}_\infty$ : $N$ est continue sur $(\K^d,\norm\cdot_\infty)$. \emph{Minoration :} la sphère $S_\infty:=\{\norm z_\infty=1\}$ est fermée et bornée dans $(\K^d,\norm\cdot_\infty)$, donc compacte (Corollaire~1.2). La fonction continue $N$ y atteint son minimum $\alpha$ (Corollaire~1.3), et $\alpha>0$ puisque $N$ ne s'annule pas sur $S_\infty$. Par homogénéité, pour $x\ne0$, $N(x)=\norm x_\infty N(x/\norm x_\infty)\ge\alpha\norm x_\infty$. Donc $\alpha\norm\cdot_\infty\le N\le\beta\norm\cdot_\infty$.

\medskip
\noindent\textbf{(c)} \emph{Young.} Soient $p,q\in\,]1,\infty[$ conjugués et $a,b\ge0$. Alors $ab\le\frac{a^p}p+\frac{b^q}q$. C'est clair si $ab=0$ ; sinon, par concavité du logarithme,
\[
\ln\Big(\frac1pa^p+\frac1qb^q\Big)\ge\frac1p\ln(a^p)+\frac1q\ln(b^q)=\ln(ab) .
\]
\emph{Hölder.} Pour $x,y\in\K^d$ et $p,q\in[1,\infty]$ conjugués, $\sum_k|x_ky_k|\le\norm x_p\norm y_q$. Si $p=1$, $q=\infty$ : $|x_ky_k|\le|x_k|\norm y_\infty$, et l'on somme. Si $1<p<\infty$ : l'inégalité est claire si $x=0$ ou $y=0$ ; sinon, par homogénéité, on peut supposer $\norm x_p=\norm y_q=1$. Young donne $|x_ky_k|\le\frac{|x_k|^p}p+\frac{|y_k|^q}q$, et en sommant, $\sum_k|x_ky_k|\le\frac1p+\frac1q=1=\norm x_p\norm y_q$.

\medskip
\noindent\textbf{(d)} Soit $u\in\ell^p$, $p<\infty$, et $t_N(u):=(u_0,\dots,u_N,0,0,\dots)\in c_{<\infty}$. Alors $\norm{u-t_N(u)}_p^p=\sum_{n>N}|u_n|^p\to0$, reste d'une série convergente : $c_{<\infty}$ est dense. Soit $D$ l'ensemble des suites à support fini et à coefficients dans $\Q$ (ou $\Q+i\Q$) ; $D$ est dénombrable, comme réunion dénombrable d'ensembles en bijection avec des $\Q^{N+1}$. Pour $u\in\ell^p$ et $\varepsilon>0$ : on choisit $N$ avec $\norm{u-t_N(u)}_p\le\frac\varepsilon2$, puis des rationnels $q_0,\dots,q_N$ avec $|u_n-q_n|\le\frac\varepsilon{2(N+1)^{1/p}}$ ; alors $v:=(q_0,\dots,q_N,0,\dots)\in D$ et $\norm{u-v}_p\le\varepsilon$.

Pour $p=\infty$ l'argument s'effondre dès la première étape : $\norm{u-t_N(u)}_\infty=\sup_{n>N}|u_n|$ ne tend vers $0$ que si $u\in c_0$. L'adhérence de $c_{<\infty}$ dans $\ell^\infty$ est $c_0$, et $\ell^\infty$ n'est pas séparable (exercice officiel~9).
\end{solution}

\begin{methode}
\textbf{Ce que ces quatre preuves ont en commun.} (a) et (b) sont des arguments d'\emph{homogénéité} : on se ramène à la sphère unité, puis on utilise soit la linéarité (a), soit la compacité de la sphère en dimension finie (b). Dans (b), l'ordre compte : d'abord la majoration $N\le\beta\norm\cdot_\infty$, qui donne la \emph{continuité} de $N$, ensuite seulement la minoration par compacité. (c) est aussi une normalisation, $\norm x_p=\norm y_q=1$. (d) est le schéma \emph{tronquer puis approcher par des rationnels}, qui marche exactement quand les queues de série tendent vers $0$ en norme, c'est-à-dire pour $p<\infty$.
\end{methode}

\begin{exo}{Exercice 8 -- Vrai ou faux ?}
Pour chaque affirmation, dire si elle est vraie ou fausse, et justifier par une preuve courte ou un contre-exemple. $E$, $F$ et $G$ désignent des EVN.
\begin{enumerate}[label=\arabic*.]
  \item Deux normes sur un même espace vectoriel sont toujours équivalentes.
  \item Si $E$ est de dimension finie, toute application linéaire $E\to F$ est continue.
  \item Tout sous-espace vectoriel de $E$ est fermé.
  \item Un sous-espace vectoriel strict de $E$ est d'intérieur vide.
  \item Si $T:E\to F$ est linéaire et bornée sur une boule $B(a,r)$, $r>0$, alors $T$ est continue.
  \item Pour $1\le p\le q\le\infty$, on a $L^p(]0,1[)\subset L^q(]0,1[)$.
  \item Dans un EVN, l'adhérence de la boule ouverte $B(a,r)$ est la boule fermée $\Bf(a,r)$.
  \item $c_{<\infty}(\N)$ est dense dans $\ell^\infty(\N)$.
  \item La norme d'un opérateur $T\in\Lc(E,F)$ est toujours atteinte sur la sphère unité de $E$.
  \item Si $T\in\Lc(E,F)$ et $S\in\Lc(F,G)$, alors $\opnorm{S\circ T}=\opnorm S\,\opnorm T$.
  \item Toute fonction continue et bornée sur $\R$ est limite uniforme sur $\R$ d'une suite de polynômes.
  \item $\Cz([0,1];\R)$ muni de la norme uniforme est séparable.
  \item Sur $\R[x]$, les normes $N(P):=\sup_{[0,1]}|P|$ et $N'(P):=\sup_{[0,2]}|P|$ sont équivalentes.
\end{enumerate}
\end{exo}

\begin{solution}
\noindent\textbf{1. Faux.} Sur $\Cz([0,1])$, $\norm f_1\le\norm f_\infty$, mais $f_n(x)=x^n$ vérifie $\norm{f_n}_\infty=1$ et $\norm{f_n}_1=\frac1{n+1}\to0$ : pas de constante $C$ avec $\norm\cdot_\infty\le C\norm\cdot_1$. C'est vrai en dimension finie (Corollaire~2.1).

\medskip
\noindent\textbf{2. Vrai.} Soit $(b_1,\dots,b_d)$ une base de $E$ et $N(x):=\max_k|x_k|$ pour $x=\sum_kx_kb_k$ ; $N$ est une norme sur $E$, équivalente à $\norm\cdot_E$ (Corollaire~2.1). Alors
\[
\norm{T(x)}_F\le\sum_k|x_k|\norm{T(b_k)}_F\le\Big(\sum_k\norm{T(b_k)}_F\Big)N(x)\le C\norm x_E .
\]

\medskip
\noindent\textbf{3. Faux.} $c_{<\infty}(\N)$ est un sous-espace dense et strict de $\ell^2(\N)$ (Proposition~2.3), donc non fermé. C'est vrai pour les sous-espaces de dimension finie (Proposition~3.2).

\medskip
\noindent\textbf{4. Vrai.} Si un sous-espace $V$ contient une boule $B(a,r)$, alors pour tout $x\ne0$, $a+\frac r{2\norm x}x\in V$ et $a\in V$, donc $x\in V$ : $V=E$.

\medskip
\noindent\textbf{5. Vrai.} Si $\norm{T(x)}\le M$ sur $B(a,r)$, alors pour $\norm z<1$, $\norm{T(z)}=\frac1r\norm{T(a+rz)-T(a)}\le\frac{2M}r$ : $T$ est bornée sur la boule unité, donc continue (Théorème~2.4). C'est le calcul final de la preuve de Banach--Steinhaus.

\medskip
\noindent\textbf{6. Faux.} C'est l'inclusion inverse : sur un ensemble de mesure finie, $L^q\subset L^p$ pour $p\le q$ (Hölder). Par exemple $x^{-1/2}\in L^1(]0,1[)\setminus L^2(]0,1[)$. L'inclusion $\ell^p\subset\ell^q$ (Proposition~2.4) va dans l'autre sens, et sur $\R$ il n'y a aucune inclusion.

\medskip
\noindent\textbf{7. Vrai.} $\Bf(a,r)$ est un fermé contenant $B(a,r)$, donc son adhérence. Réciproquement, si $\norm{x-a}=r$, les points $x_n:=a+(1-\frac1n)(x-a)$ sont dans $B(a,r)$ et convergent vers $x$. C'est la structure vectorielle (les homothéties) qui rend vrai ce qui est faux dans un espace métrique général (TD1, Exercice~8, question~1).

\medskip
\noindent\textbf{8. Faux.} Pour $v\in c_{<\infty}$, $\norm{\mathbf 1-v}_\infty\ge1$, puisque $v_n=0$ pour $n$ grand. L'adhérence de $c_{<\infty}$ dans $\ell^\infty$ est $c_0$ (Exercice~5). C'est vrai dans $\ell^p$ pour $p<\infty$ (Proposition~2.3).

\medskip
\noindent\textbf{9. Faux.} Sur $c_0(\N)$, $f(u):=\sum_n2^{-n-1}u_n$ est de norme $1$, non atteinte (Exercice~3(d)). C'est vrai si $E$ est de dimension finie : la sphère unité est compacte et $x\mapsto\norm{Tx}$ est continue.

\medskip
\noindent\textbf{10. Faux.} On a seulement $\opnorm{S\circ T}\le\opnorm S\,\opnorm T$. Pour $T=S=\begin{pmatrix}0&1\\0&0\end{pmatrix}$ sur $\R^2$, $T^2=0$ alors que $\opnorm T\ne0$.

\medskip
\noindent\textbf{11. Faux.} Si des polynômes $P_n$ convergent uniformément sur $\R$, alors pour $n,m$ grands $P_n-P_m$ est un polynôme borné sur $\R$, donc constant : $P_n=P_N+c_n$ pour $n\ge N$, et la limite est le polynôme $P_N+\lim c_n$. Or un polynôme borné sur $\R$ est constant : $\sin$ n'est pas limite uniforme de polynômes sur $\R$. Le théorème de Weierstrass demande un \emph{compact}.

\medskip
\noindent\textbf{12. Vrai.} Les polynômes à coefficients rationnels forment une partie dénombrable, dense dans les polynômes pour $\norm\cdot_\infty$ sur $[0,1]$ (approcher chaque coefficient), eux-mêmes denses par Weierstrass (Exemple~2.3 ; exercice officiel~5 pour un compact métrique quelconque).

\medskip
\noindent\textbf{13. Faux.} $N\le N'$, mais $P_n:=(x/2)^n$ vérifie $N'(P_n)=1$ et $N(P_n)=2^{-n}\to0$ : pas de constante $C$ avec $N'\le C\,N$. Sur $\R_d[x]$, de dimension finie, elles sont équivalentes (Corollaire~2.1), avec une constante qui explose quand $d\to\infty$.
\end{solution}

\begin{piege}
\textbf{La dimension finie, hypothèse cachée du chapitre~2.} En dimension finie tout est vrai : normes équivalentes, applications linéaires continues, sous-espaces fermés, normes d'opérateur atteintes (1, 2, 3, 9, 13). En dimension infinie il faut à chaque fois préciser \emph{quelle norme}, et les contre-exemples vivent dans $c_{<\infty}$, $c_0$, $\R[x]$ et $(\Cz([0,1]),\norm\cdot_1)$. Deuxième famille de pièges : le sens des inclusions entre espaces $\ell^p$ et $L^p$ (6) et le rôle de l'exposant $p=\infty$, pour lequel densité de $c_{<\infty}$ et séparabilité tombent (8). Troisième : la compacité dans Weierstrass (11, 12).
\end{piege}
